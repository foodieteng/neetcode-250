完整實作// SORT + REUSE + BREAK

p39.cpp C++17 
class Solution {
public:
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        sort(candidates.begin(), candidates.end());         // break 的前提
        vector<vector<int>> ans;
        vector<int> path;
        dfs(candidates, ans, path, 0, target, 0);
        return ans;
    }

    void dfs(vector<int>& candidates, vector<vector<int>>& ans, vector<int> path, int res, int target, int s) {
        if (res == target) {                                // 湊到了
            ans.push_back(path);
            return;
        }

        for (int i = s; i < candidates.size(); i++) {
            if (res + candidates[i] > target) break;        // 後面只會更大

            path.push_back(candidates[i]);
            dfs(candidates, ans, path, res + candidates[i], target, i);   // i:可以再選自己
            path.pop_back();
        }
    }
};

三個參數,三件事:i 管「能不能重複選」、s 起點管「會不會順序重複」、sort 管「break 會不會漏」。 官方範例 1 的 candidates 本來就是排好的,「沒排序」的錯在範例裡看不出來。
