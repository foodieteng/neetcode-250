完整實作 · res 傳值(我的)// STRING BY VALUE

p22.cpp C++17 
class Solution {
public:
    vector<string> generateParenthesis(int n) {
        vector<string> ans;
        string res = "";
        dfs(ans, res, 0, 0, n);
        return ans;
    }

    void dfs(vector<string>& ans, string res, int open, int close, int n) {
        if (res.size() == 2 * n) {                  // 長度到 2n:一定合法,收
            ans.push_back(res);
            return;
        }

        if (open < n) {                             // '(' 還沒用完
            dfs(ans, res + '(', open + 1, close, n);
        }

        if (close < open) {                          // 還有沒配對的 '('
            dfs(ans, res + ')', open, close + 1, n);
        }
    }
};

    void dfs(vector<string>& ans, string& res, int open, int close, int n) {
        if (res.size() == 2 * n) {
            ans.push_back(res);                      // 複製一份存起來
            return;
        }

        if (open < n) {
            res.push_back('(');
            dfs(ans, res, open + 1, close, n);
            res.pop_back();                          // 一定要改回來
        }

        if (close < open) {
            res.push_back(')');
            dfs(ans, res, open, close + 1, n);
            res.pop_back();
        }
    }
