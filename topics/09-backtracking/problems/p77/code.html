p77.cpp C++17 
class Solution {
public:
    vector<vector<int>> combine(int n, int k) {
        vector<vector<int>> ans;
        vector<int> path;
        dfs(ans, path, n, k, 1);
        return ans;
    }

    void dfs(vector<vector<int>>& ans, vector<int>& path, int n, int k, int s) {
        if (path.size() == k) {                             // 選滿 k 個才收
            ans.push_back(path);
            return;
        }

        for (int i = s; i <= n; i++) {                      // 數字是 1..n
            path.push_back(i);
            dfs(ans, path, n, k, i + 1);                    // 每個數最多一次
            path.pop_back();
        }
    }
};

p77_pruned.cpp C++17 
class Solution {
public:
    vector<vector<int>> combine(int n, int k) {
        vector<vector<int>> ans;
        vector<int> path;
        dfs(ans, path, n, k, 1);
        return ans;
    }

    void dfs(vector<vector<int>>& ans, vector<int>& path, int n, int k, int s) {
        if (path.size() == k) {
            ans.push_back(path);
            return;
        }

        int need = k - path.size();                         // 還差幾個
        for (int i = s; i <= n - need + 1; i++) {           // i..n 至少要有 need 個
            path.push_back(i);
            dfs(ans, path, n, k, i + 1);
            path.pop_back();
        }
    }
};

邊界與陷阱// PITFALLS

改法(55 組窮舉)實測最小反例
剪枝上界少了 +1:i <= n - need全錯n=1, k=1 → []
迴圈從 0 開始(數字是 1..n)全錯n=1, k=1 → [[0],[1]]
條件寫 size() > k全錯n=1, k=1 → []
遞迴傳 s + 145n=2, k=2 → [[1,2],[2,2]]
遞迴傳 i(39 的寫法)45n=2, k=2 → [[1,1],[1,2],[2,2]]
收完沒 return0 錯多跑:總呼叫 18,434 vs 11,253
k = 1 的錯會讓每一組都錯,k = 1 時答案都對的錯(s + 1、傳 i)要 k ≥ 2 才看得出來。 測資至少放一組 n = 1, k = 1 和一組 n = 2, k = 2。
