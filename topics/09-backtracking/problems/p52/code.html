完整實作// COUNT AT THE LEAF

p52.cpp C++17 
class Solution {
public:
    int totalNQueens(int n) {
        int ans = 0;
        vector<bool> row(n, false), col(n, false), diag1(2 * n - 1, false), diag2(2 * n - 1, false);
        dfs(n, 0, ans, row, col, diag1, diag2);
        return ans;
    }

    void dfs(int n, int r, int& ans, vector<bool>& row,  vector<bool>& col, vector<bool>& diag1, vector<bool>& diag2) {
        if (r == n) {                                       // ① 每一列都放好 → 數一個
            ans += 1;
            return;
        }

        for (int c = 0; c < n; c++) {                       // ② 第 r 列試每一欄
            if (row[r] || col[c] || diag1[r + c] || diag2[r - c + n - 1]) continue;   // ③ 被佔就跳過

            row[r] = col[c] = diag1[r + c] = diag2[r - c + n - 1] = true;           // ④ 放
            dfs(n, r + 1, ans, row, col, diag1, diag2);
            row[r] = col[c] = diag1[r + c] = diag2[r - c + n - 1] = false;          // ⑤ 拿掉
        }
    }
};

Bitmask 計數版 · 再加左右對稱// BITMASK + MIRROR

和 51 程式碼頁第 2 節同一招:三個 int 遮罩,一次算出這一列所有能放的欄。只要個數,所以連 queenCol 都不用記,dfs 直接回傳子樹的解數。

p52_bitmask.cpp C++17 
class Solution {
public:
    int totalNQueens(int n) {
        fullMask = (1 << n) - 1;
        return dfs(0, 0, 0);
    }

private:
    int fullMask = 0;

    int dfs(int usedCols, int mainDiag, int antiDiag) {
        if (usedCols == fullMask) return 1;                          // 每一欄都有皇后 = n 列都放好
        int count = 0;
        int available = ~(usedCols | mainDiag | antiDiag) & fullMask;
        while (available) {
            int bit = available & -available;
            available ^= bit;
            count += dfs(usedCols | bit, ((mainDiag | bit) << 1) & fullMask, (antiDiag | bit) >> 1);
        }
        return count;
    }
};

再用概念頁 LEMMA 2 的鏡像:第 0 列只試左半邊,結果 × 2;n 為奇數時正中間那欄單獨算。dfs 完全不改,只改入口。

p52_mirror.cpp C++17 
class Solution {
public:
    int totalNQueens(int n) {
        fullMask = (1 << n) - 1;
        int total = 0;
        for (int c = 0; c < n / 2; c++) {                            // 左半邊:每個解都有一個鏡像
            int bit = 1 << c;
            total += 2 * dfs(bit, (bit << 1) & fullMask, bit >> 1);
        }
        if (n % 2 == 1) {                                            // 正中間那欄:鏡像是自己
            int bit = 1 << (n / 2);
            total += dfs(bit, (bit << 1) & fullMask, bit >> 1);
        }
        return total;
    }

private:
    int fullMask = 0;

    int dfs(int usedCols, int mainDiag, int antiDiag) {
        if (usedCols == fullMask) return 1;
        int count = 0;
        int available = ~(usedCols | mainDiag | antiDiag) & fullMask;
        while (available) {
            int bit = available & -available;
            available ^= bit;
            count += dfs(usedCols | bit, ((mainDiag | bit) << 1) & fullMask, (antiDiag | bit) >> 1);
        }
        return count;
    }
};

細節走讀// WALKTHROUGH · n = 4

和概念頁動畫同一組,插樁實跑。每行最後的 → k 是那個節點底下數到的解數(= 回到它時 ans 增加了多少)。
