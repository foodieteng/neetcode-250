完整實作// MARK · EXPLORE · RESTORE

p79.cpp C++17 
class Solution {
public:
    const vector<pair<int, int>> dirs = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};   // 上 下 左 右

    bool exist(vector<vector<char>>& board, string word) {
        int m = board.size(), n = board[0].size();
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (dfs(board, word, i, j, 0)) return true;   // 每一格都當起點
            }
        }

        return false;
    }

    bool dfs(vector<vector<char>>& board, string& word, int r, int c, int i) {
        if (word[i] != board[r][c]) return false;      // ① 先比字元
        if (i == word.size() - 1) return true;        // ② 最後一個也對了

        const char cache = board[r][c];
        board[r][c] = '#';                              // ③ 標:這條路用過了

        for (auto [dr, dc] : dirs) {
            auto nr = r + dr, nc = c + dc;
            if (nr < 0 || nr >= board.size() || nc < 0 || nc >= board[0].size() ||
                board[nr][nc] == '#') continue;   // 出界或在路徑上
            if (dfs(board, word, nr, nc, i + 1)) return true;   // ④ 成功:直接往上傳
        }

        board[r][c] = cache;                             // ⑤ 全失敗:還原
        return false;
    }
};

Follow-up 剪枝 · 先數字母、從稀有端開始// PRUNING FOR LARGER BOARDS

dfs 完全不動,只在 exist 開頭加兩件事(Follow-up 問的「更大的 board 怎麼剪枝」)。

p79_pruned.cpp C++17 
class Solution {
public:
    const vector<pair<int, int>> dirs = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

    bool exist(vector<vector<char>>& board, string word) {
        int boardCount[128] = {0}, wordCount[128] = {0};
        for (auto& row : board)
            for (char ch : row) boardCount[ch]++;
        for (char ch : word)
            if (++wordCount[ch] > boardCount[ch]) return false;   // ① 某個字母 board 上不夠 → 不用搜
        if (boardCount[word.back()] < boardCount[word[0]])
            reverse(word.begin(), word.end());                    // ② 從比較稀有的那一端開始找

        int m = board.size(), n = board[0].size();
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (dfs(board, word, i, j, 0)) return true;
            }
        }

        return false;
    }

    bool dfs(vector<vector<char>>& board, string& word, int r, int c, int i) {
        if (word[i] != board[r][c]) return false;
        if (i == word.size() - 1) return true;

        const char cache = board[r][c];
        board[r][c] = '#';

        for (auto [dr, dc] : dirs) {
            auto nr = r + dr, nc = c + dc;
            if (nr < 0 || nr >= board.size() || nc < 0 || nc >= board[0].size() ||
                board[nr][nc] == '#') continue;
            if (dfs(board, word, nr, nc, i + 1)) return true;
        }

        board[r][c] = cache;
        return false;
    }
};

兩招為什麼成立

① 數字母:word 裡某個字母出現的次數比 board 上多,同一格又只能用一次 → 一定拼不出來,直接 false。最壞例子的 board 根本沒有 B,一次 dfs 都不用跑。 ② 反過來找:一條路徑倒過來走還是合法路徑,所以找 word 和找 reverse(word) 答案一樣。從 board 上比較少的字母開始,能通過第一個字元比較的起點變少,整棵搜尋樹就小很多(上表第二列:B 開頭只要 36 次)。 word 是傳值進來的,反轉不會影響呼叫的人。
