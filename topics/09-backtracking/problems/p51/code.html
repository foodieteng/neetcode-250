完整實作// ROW BY ROW

p51.cpp C++17 
class Solution {
public:
    vector<vector<string>> solveNQueens(int n) {
        vector<vector<string>> ans;
        vector<string> board(n, string(n, '.'));
        vector<bool> row(n, false), col(n, false), diag1(2 * n - 1, false), diag2(2 * n - 1, false);
        dfs(ans, board, n, 0, row, col, diag1, diag2);
        return ans;
    }

    void dfs(vector<vector<string>>& ans, vector<string>& board, int n, int r, vector<bool>& row, vector<bool>& col, vector<bool>& diag1, vector<bool>& diag2) {
        if (r >= n) {                                        // ① 每一列都放好了
            ans.push_back(board);
            return;
        }

        for (int c = 0; c < n; c++) {                        // ② 第 r 列試每一欄
            if (col[c] || diag1[r + c] || diag2[r - c + n - 1]) continue;   // ③ 欄、/、\ 被佔

            board[r][c] = 'Q';                               // ④ 放
            row[r] = col[c] = diag1[r + c] = diag2[r - c + n - 1] = true;
            dfs(ans, board, n, r + 1, row, col, diag1, diag2);
            row[r] = col[c] = diag1[r + c] = diag2[r - c + n - 1] = false;   // ⑤ 拿掉
            board[r][c] = '.';
        }
    }
};

Bitmask 優化版// BITMASK · ABOUT 5–6× FASTER

把三個 vector<bool> 換成三個 int:第 c 個 bit = 第 c 欄被佔。這一列所有能放的欄一次算出來,不用一格一格試再 continue。

p51_bitmask.cpp C++17 
class Solution {
public:
    vector<vector<string>> solveNQueens(int n) {
        size = n;
        fullMask = (1 << n) - 1;
        queenCol.assign(n, 0);
        result.clear();
        dfs(0, 0, 0, 0);
        return result;
    }

private:
    int size = 0, fullMask = 0;
    vector<int> queenCol;                 // queenCol[r] = 第 r 列皇后的欄
    vector<vector<string>> result;

    // usedCols:被佔的欄;mainDiag:"\" 斜線(下一列往右移 → <<1);antiDiag:"/" 斜線(下一列往左移 → >>1)
    void dfs(int row, int usedCols, int mainDiag, int antiDiag) {
        if (row == size) {                                   // 每一列都放好 → 這時才建棋盤
            vector<string> board(size, string(size, '.'));
            for (int r = 0; r < size; r++) board[r][queenCol[r]] = 'Q';
            result.push_back(move(board));
            return;
        }
        int available = ~(usedCols | mainDiag | antiDiag) & fullMask;   // 可放的欄
        while (available) {
            int bit = available & -available;                // 取最低位(欄由小到大)
            available ^= bit;
            queenCol[row] = __builtin_ctz(bit);
            dfs(row + 1, usedCols | bit, ((mainDiag | bit) << 1) & fullMask, (antiDiag | bit) >> 1);
        }
    }
};

為什麼快:遞迴次數一樣(n = 9 都是 8,394 次),省的是每一次裡面的工作

① 不用試被擋的格子:~(usedCols | mainDiag | antiDiag) & fullMask 一個運算得到這列所有可放的欄;你的版本在 n = 9 要做 63,985 次「查三個陣列 → continue」。 ② 斜線不用算索引:「\」往下一列會往右一欄,所以整個 mask << 1;「/」往左一欄,>> 1。不用 r + c、r - c + n - 1,也不用撤銷 —— 三個 mask 是傳值,和 17 的 res 一樣。 ③ 棋盤最後才建:路上只記 queenCol[r],到葉子才畫出 n 個字串。

正確性:n = 1~12 和你的版本輸出完全相同,連順序都一樣(available & -available 取最低位 = 欄由小到大);ASan 乾淨。 要記的三行:available = ~(cols | d1 | d2) & full、bit = available & -available、遞迴傳 (cols | bit, (d1 | bit) << 1, (d2 | bit) >> 1)。
