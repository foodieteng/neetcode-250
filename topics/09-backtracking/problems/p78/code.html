完整實作 · path 傳值(我的)// PATH BY VALUE

p78.cpp C++17 
class Solution {
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> ans;
        vector<int> path;
        dfs(nums, ans, path, 0);
        return ans;
    }

    void dfs(vector<int>& nums, vector<vector<int>>& ans, vector<int> path, int s) {
        ans.push_back(path);                                // 每個節點都是一個子集

        for (int i = s; i < nums.size(); i++) {
            path.push_back(nums[i]);                        // 選
            dfs(nums, ans, path, i + 1);                    // 傳一份複本下去
            path.pop_back();                                // 撤銷,換下一個
        }
    }
};

class Solution {
public:
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> ans;
        vector<int> path;
        dfs(nums, ans, path, 0);
        return ans;
    }

    void dfs(vector<int>& nums, vector<vector<int>>& ans, vector<int>& path, int s) {
        ans.push_back(path);                                // 複製一份存起來

        for (int i = s; i < nums.size(); i++) {
            path.push_back(nums[i]);                        // 改共用的 path
            dfs(nums, ans, path, i + 1);
            path.pop_back();                                // 一定要改回來
        }
    }
};

邊界與陷阱// PITFALLS

改法(3,619 組窮舉)實測最小反例
傳參考,沒有 pop_back3,612[1,2] → [[],[1],[1,2],[1,2,2]]
傳值,沒有 pop_back3,612[1,2] → [[],[1],[1,2],[1,2]](該是 [2])
遞迴傳 s + 1 不是 i + 13,612[1,2] → [[],[1],[1,2],[2],[2,2]]
只在 s == n 時收(混進二元寫法)全錯[1] → [[1]],少了空集合
ans.push_back(path) 移到迴圈後面0 錯順序變了(先深後淺),內容一樣
兩種「沒 pop」錯得不一樣:傳參考時多出來的元素會跟著往下傳([1,2,2]); 傳值時下一層拿到的是複本,所以只錯在同一層的兄弟([1,2] 出現兩次)。 只有 1 個數的輸入每一種錯都抓不到 —— 至少要測兩個數。

細節走讀// WALKTHROUGH

[1,2,3](和概念頁動畫同一組),實跑印出的 push / pop 順序。

// PHASE 1: 從 1 開頭的子集 call#1 s=0 path [] → ans #1 push 1 → [1] call#2 s=1 path [1] → ans #2 push 2 → [1,2] call#3 s=2 path [1,2] → ans #3 push 3 → [1,2,3] call#4 s=3 path [1,2,3] → ans #4 迴圈不跑,return pop → [1,2] pop → [1] push 3 → [1,3] call#5 s=3 path [1,3] → ans #5 pop → [1] pop → []
// PHASE 2: 從 2、3 開頭的子集 push 2 → [2] call#6 s=2 path [2] → ans #6 push 3 → [2,3] call#7 s=3 path [2,3] → ans #7 pop → [2] pop → [] push 3 → [3] call#8 s=3 path [3] → ans #8 pop → [] // OUTPUT: [[],[1],[1,2],[1,2,3],[1,3],[2],[2,3],[3]]
PREV概念與動畫 NEXT觀念題與走讀

ans.push_back(path) 放最前面;迴圈 push → dfs(i + 1) → pop。path 傳值也對,傳參考少一次複製(n = 20:122 vs 104 ms)。沒 pop、s + 1、只在葉子收都會錯;測資至少要兩個數。
