class Solution {
public:
    vector<vector<string>> partition(string s) {
        vector<vector<string>> ans;
        vector<string> path;
        dfs(ans, path, s, 0);
        return ans;
    }

    bool isPalindrome(string s, int start, int end) {      // 傳值:每次複製一份 s
        while (start < end) {
            if (s[start] != s[end]) return false;
            start++; end--;
        }
        return true;
    }

    void dfs(vector<vector<string>>& ans, vector<string>& path, string& s, int start) {
        if (start >= s.size()) {                           // ① 切到底:收答案
            ans.push_back(path);
            return;
        }

        for (int end = start; end < s.size(); end++) {     // ② 第一段 = s[start..end]
            if (isPalindrome(s, start, end)) {             // ③ 不是回文就換下一個 end
                int L = end - start + 1;
                path.push_back(s.substr(start, L));        // ④ 切下來
                dfs(ans, path, s, end + 1);                //    剩下的交給下一層
                path.pop_back();                           // ⑤ 撤銷
            }
        }
    }
};

改法窮舉錯最小反例

和概念頁動畫同一組,插樁實跑印出來的順序。#k = 第 k 次 dfs。

// PHASE 1: 第一段 = "a"(end = 0) #1 dfs(0) path = [] end=0 "a" 是回文 → push [a] → #2 dfs(1) end=1 "a" 是回文 → push [a,a] → #3 dfs(2) end=2 "b" 是回文 → push [a,a,b] → #4 dfs(3):start = 3 = size,收 ans#1 [a,a,b] pop → [a,a];#3 迴圈結束 pop → [a] end=2 "ab" 不是回文 → 換下一個 end(已經沒有) pop → []
// PHASE 2: 第一段 = "aa"(end = 1),再試 "aab"(end = 2) end=1 "aa" 是回文 → push [aa] → #5 dfs(2) end=2 "b" 是回文 → push [aa,b] → #6 dfs(3):收 ans#2 [aa,b] pop → [aa] pop → [] end=2 "aab" 不是回文 // OUTPUT: [[a,a,b],[aa,b]](6 次 dfs、7 次回文檢查)
