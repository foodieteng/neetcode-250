完整實作// SORT + SAME-LEVEL SKIP + BREAK

p40.cpp C++17 
class Solution {
public:
    vector<vector<int>> combinationSum2(vector<int>& candidates, int target) {
        sort(candidates.begin(), candidates.end());         // 相同的數排在一起
        vector<vector<int>> ans;
        vector<int> path;
        dfs(candidates, ans, path, target, 0, 0);
        return ans;
    }

    void dfs(vector<int>& nums, vector<vector<int>>& ans, vector<int> path, int target, int cur, int s) {
        if (cur == target) {
            ans.push_back(path);
            return;
        }

        for (int i = s; i < nums.size(); i++) {
            if (i > s && nums[i] == nums[i - 1]) continue;  // 同一層:相同的值只試第一個
            if (cur + nums[i] > target) break;              // 排好序,後面只會更大

            path.push_back(nums[i]);
            dfs(nums, ans, path, target, cur + nums[i], i + 1);   // i + 1:每個位置只用一次
            path.pop_back();
        }
    }
};

邊界與陷阱// PITFALLS

改法(8,736 組窮舉)實測最小反例
去重寫 i > 05,735[1,1], 2 → [](正解 [[1,1]])
沒有去重那一行7,343[1,1], 1 → [[1],[1]]
遞迴傳 i(39 的寫法)5,985[1], 2 → [[1,1]],同一個 1 用了兩次
沒排序5,961[2,1], 1 → [](在 2 就 break)
兩個 if 對調(先 break 再去重)0 錯相同的值大小一樣,誰先判斷都對
不跳,最後丟 set0 錯對,但 25 個 1 要 7,119,516 次呼叫
沒排序,去重那一行也會失效:nums[i] == nums[i-1] 只比相鄰的,相同的值沒排在一起就抓不到。 這題的 sort 同時撐著兩件事:break 和去重。
