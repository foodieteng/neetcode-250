完整實作// FOUR BUCKETS

p473.cpp C++17 
class Solution {
public:
    bool makesquare(vector<int>& matchsticks) {
        sort(matchsticks.begin(), matchsticks.end(), greater<int>());   // 大的先放
        int n = matchsticks.size();
        int total = accumulate(matchsticks.begin(), matchsticks.end(), 0);
        if (total % 4 != 0) return false;
        int target = total / 4;
        vector<int> buckets(4, 0);

        return exist(matchsticks, buckets, target, 0);
    }

    bool exist(vector<int>& nums, vector<int>& buckets, int target, int idx) {
        if (idx >= nums.size()) return true;                  // ① 每根都放好了

        for (int i = 0; i < 4; i++) {                         // ② 這根試 4 個邊
            if (buckets[i] + nums[idx] > target) continue;    // ③ 放不下
            if (i > 0 && buckets[i] == buckets[i - 1]) continue;   // ④ 和左邊同長:對稱

            buckets[i] += nums[idx];
            if (exist(nums, buckets, target, idx + 1)) return true;
            buckets[i] -= nums[idx];                          // ⑤ 拿回來
        }

        return false;
    }
};
