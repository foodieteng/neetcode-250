p140.cpp C++17 
class Solution {
public:
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        unordered_set<string> dict(wordDict.begin(), wordDict.end());
        int n = s.size();
        maxLen = 0;
        for (const string& word : wordDict) maxLen = max(maxLen, (int)word.size());

        // canBreak[i]:s[i..n) 能不能完整切成字典裡的字
        canBreak.assign(n + 1, false);
        canBreak[n] = true;
        for (int i = n - 1; i >= 0; i--) {
            for (int end = i + 1; end <= min(n, i + maxLen); end++) {
                if (canBreak[end] && dict.count(s.substr(i, end - i))) {
                    canBreak[i] = true;
                    break;
                }
            }
        }

        vector<string> ans;
        if (!canBreak[0]) return ans;                               // 整串切不完
        string path;
        dfs(s, 0, dict, path, ans);
        return ans;
    }

private:
    int maxLen = 0;
    vector<bool> canBreak;

    void dfs(const string& s, int start, unordered_set<string>& dict, string& path, vector<string>& ans) {
        int n = s.size();
        if (start == n) {
            ans.push_back(path);
            return;
        }

        for (int end = start + 1; end <= min(n, start + maxLen); end++) {
            if (!canBreak[end]) continue;                           // 切到這裡之後走不到底
            string word = s.substr(start, end - start);
            if (!dict.count(word)) continue;

            int oldLen = path.size();
            if (!path.empty()) path += ' ';
            path += word;
            dfs(s, end, dict, path, ans);
            path.resize(oldLen);
        }
    }
};

① 純回溯:最佳解拿掉 canBreak 和 maxLen 就是它。面試時先寫這個最快,再補剪枝。

p140_backtrack.cpp C++17 
class Solution {
public:
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        unordered_set<string> dict(wordDict.begin(), wordDict.end());
        vector<string> ans;
        string path;
        dfs(s, 0, dict, path, ans);
        return ans;
    }

    void dfs(const string& s, int start, unordered_set<string>& dict, string& path, vector<string>& ans) {
        if (start == (int)s.size()) {                               // ① 整串切完
            ans.push_back(path);
            return;
        }

        for (int end = start + 1; end <= (int)s.size(); end++) {    // ② 這個字切到哪
            string word = s.substr(start, end - start);
            if (!dict.count(word)) continue;                        // ③ 不是字

            int oldLen = path.size();                               // ④ 接上
            if (!path.empty()) path += ' ';
            path += word;
            dfs(s, end, dict, path, ans);
            path.resize(oldLen);                                    // ⑤ 拿掉(連空格)
        }
    }
};

② 記憶化 DFS:dfs(start) 不再往下傳 path,而是回傳 s[start..n) 的所有句子;父層把自己切下的字接在每一句前面。同一個 start 只算一次。

p140_memo.cpp C++17 
class Solution {
public:
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        unordered_set<string> dict(wordDict.begin(), wordDict.end());
        unordered_map<int, vector<string>> memo;
        return dfs(s, 0, dict, memo);
    }

    // 回傳 s[start..n) 的所有句子
    vector<string> dfs(const string& s, int start, unordered_set<string>& dict, unordered_map<int, vector<string>>& memo) {
        auto it = memo.find(start);
        if (it != memo.end()) return it->second;                    // ① 算過了
        if (start == (int)s.size()) return {""};                    // ② 空後綴 = 一句空句子

        vector<string> sentences;
        for (int end = start + 1; end <= (int)s.size(); end++) {
            string word = s.substr(start, end - start);
            if (!dict.count(word)) continue;
            for (const string& rest : dfs(s, end, dict, memo)) {    // ③ 接在每個後綴句子前面
                sentences.push_back(rest.empty() ? word : word + " " + rest);
            }
        }
        memo[start] = sentences;                                    // ④ 存起來(可能是空的)
        return sentences;
    }
};
base case 為什麼是 {""} 而不是 { }

{""} 是「一句空句子」:空後綴有一種切法(什麼都不切)。父層拿到它,接上自己的字就變成 "dog"。 { } 是「零種切法」—— 寫錯成它,每一層都拿到空清單,所有輸入都回傳 [](實測 4,307 / 20,000 組錯)。 同理 rest.empty() ? word : word + " " + rest 是在擋最後一個字後面多一個空格。
