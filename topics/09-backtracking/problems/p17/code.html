完整實作// PICK ONE LETTER PER DIGIT

p17.cpp C++17 
class Solution {
private:
    const vector<string> mp = {"", "", "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"};

public:
    vector<string> letterCombinations(string digits) {
        vector<string> ans;
        string res = "";
        dfs(ans, res, digits, 0);
        return ans;
    }

    void dfs(vector<string>& ans, string res, string& digits, int i) {   // res 傳值
        if (i >= digits.size()) {                    // ① 每位都選好了
            ans.push_back(res);
            return;
        }

        for (char c : mp[digits[i] - '0']) {         // ② 第 i 位的每個字母
            dfs(ans, res + c, digits, i + 1);        // ③ 新字串給下一層,不用撤銷
        }
    }
};
