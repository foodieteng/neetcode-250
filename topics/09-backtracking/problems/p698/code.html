完整實作// K BUCKETS

p698.cpp C++17 
class Solution {
public:
    bool canPartitionKSubsets(vector<int>& nums, int k) {
        sort(nums.begin(), nums.end(), greater<int>());       // 大的先放
        int total = accumulate(nums.begin(), nums.end(), 0);
        if (total % k != 0) return false;
        int target = total / k;
        vector<int> buckets(k, 0);

        return exist(nums, buckets, target, k, 0);
    }

    bool exist(vector<int>& nums, vector<int>& buckets, int target, int k, int idx) {
        if (idx >= nums.size()) return true;                  // ① 每個數都放好了

        for (int i = 0; i < k; i++) {                         // ② 這個數試 k 個桶
            if (buckets[i] + nums[idx] > target) continue;    // ③ 放不下
            if (i > 0 && buckets[i] == buckets[i - 1]) continue;   // ④ 和左邊同高:對稱

            buckets[i] += nums[idx];
            if (exist(nums, buckets, target, k, idx + 1)) return true;
            buckets[i] -= nums[idx];                          // ⑤ 拿回來
        }

        return false;
    }
};
