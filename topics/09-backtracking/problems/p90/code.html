完整實作// SORT + SAME-LEVEL SKIP

p90.cpp C++17 
class Solution {
public:
    vector<vector<int>> subsetsWithDup(vector<int>& nums) {
        sort(nums.begin(), nums.end());                     // 相同的值排在一起
        vector<vector<int>> ans;
        vector<int> path;
        dfs(nums, ans, path, 0);
        return ans;
    }

    void dfs(vector<int>& nums, vector<vector<int>>& ans, vector<int> path, int s) {
        ans.push_back(path);                                // 每個節點都是一個子集

        for (int i = s; i < nums.size(); i++) {
            if (i > s && nums[i] == nums[i - 1]) continue;  // 同一層:相同的值只試第一個

            path.push_back(nums[i]);
            dfs(nums, ans, path, i + 1);
            path.pop_back();
        }
    }
};

邊界與陷阱// PITFALLS

改法(3,279 組窮舉)實測最小反例
