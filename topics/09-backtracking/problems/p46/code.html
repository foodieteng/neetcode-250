class Solution {
public:
    vector<vector<int>> permute(vector<int>& nums) {
        int n = nums.size();
        vector<vector<int>> ans;
        vector<int> path;
        vector<bool> used(n, false);
        dfs(nums, ans, path, used);
        return ans;
    }

    void dfs(vector<int>& nums, vector<vector<int>>& ans, vector<int>& path, vector<bool>& used) {
        if (path.size() == nums.size()) {                   // 每一格都填了
            ans.push_back(path);
            return;
        }

        for (int i = 0; i < nums.size(); i++) {             // 每一格都可以從頭挑
            if (used[i]) continue;                          // 已經在 path 裡

            used[i] = true;
            path.push_back(nums[i]);
            dfs(nums, ans, path, used);
            path.pop_back();                                // 兩個都要還原
            used[i] = false;
        }
    }
};
