完整實作// POST-ORDER

p1325.cpp C++17 
class Solution {
public:
    TreeNode* removeLeafNodes(TreeNode* root, int target) {
        if (!root)
            return nullptr;

        // ① 先清左右子樹,並把結果「接回來」(被刪的會回 nullptr)
        root->left = removeLeafNodes(root->left, target);
        root->right = removeLeafNodes(root->right, target);

        // ② 這時看到的是「最終」的小孩:三個條件都成立才刪自己
        return (!root->left && !root->right && root->val == target) ? nullptr
                                                                    : root;
    }
};

// PHASE 1: 左子樹(B 那一支) D(2) 左空 ✓ 右空 ✓ val==2 ✓ → return nullptr ← B->left = nullptr B(2) 左空 ✓ 右空 ✓ val==2 ✓ → return nullptr ← A->left = nullptr ★ B 原本有小孩 D,不是葉子;D 被切掉後它「變成」葉子 —— 連鎖刪除
// PHASE 2: 右子樹(C 那一支) E(2) 左空 ✓ 右空 ✓ val==2 ✓ → return nullptr ← C->left = nullptr F(4) 左空 ✓ 右空 ✓ val==2 ✗ → return F ← C->right = F C(3) 左空 ✓ 右空 ✗(有 F) → return C ← A->right = C
// PHASE 3: 根 A(1) 左空 ✓ 右空 ✗(有 C) → return A 總呼叫次數 13 = 2·6 + 1(6 個真節點 + 7 個空指標) // OUTPUT: [1,null,3,null,4]
