One thing worth flagging — the propagation is only correct because I process in topological order. When I pop u, every prerequisite of u has already been processed, so the whole column table[i][u] is final. That's exactly what lets me copy it forward to v without missing anything.

class Solution {
public:
    vector<bool> checkIfPrerequisite(int numCourses,
            vector<vector<int>>& prerequisites, vector<vector<int>>& queries) {
        vector<vector<bool>> table(numCourses, vector<bool>(numCourses, false));
        vector<vector<int>> graph(numCourses);
        vector<int> indeg(numCourses, 0);

        for (auto& pre : prerequisites) {
            int a = pre[0], b = pre[1];
            graph[a].push_back(b);              // edge a→b (a is a prerequisite of b)
            indeg[b]++;
        }

        queue<int> q;
        for (int i = 0; i < numCourses; i++)
            if (indeg[i] == 0) q.push(i);

        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : graph[u]) {
                table[u][v] = true;                // direct: u is a prerequisite of v
                if (--indeg[v] == 0) q.push(v);

                for (int i = 0; i < numCourses; i++) {
                    if (i == u || !table[i][u]) continue;
                    table[i][v] = true;            // transitive: i before u → i before v
                }
            }
        }

        vector<bool> ans;
        for (auto& qy : queries) ans.push_back(table[qy[0]][qy[1]]);   // O(1) lookup
        return ans;
    }
};

I build the edges a → b with in-degrees, seed the queue with the zero in-degree courses, and pop them in order. For each edge u → v, I set table[u][v] to true, decrement v's in-degree, and then loop over every course i: if i is a prerequisite of u, it becomes a prerequisite of v too. At the end each query is a single table lookup.

One thing worth flagging — that inner for i is where the O(V) factor comes from: for each edge I copy up to V prerequisites forward. That's the whole cost of building all-pairs reachability.
