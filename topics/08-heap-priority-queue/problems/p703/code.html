完整實作// MIN-HEAP OF SIZE K

p703.cpp C++17 
class KthLargest {
public:
    // 最小堆:greater 讓最小的在 top —— 堆頂 = 前 k 名裡最小的 = 第 k 大
    priority_queue<int, vector<int>, greater<int>> pq;
    int size;
    KthLargest(int k, vector<int>& nums) {
        size = k;
        for (int num : nums) {
            pq.push(num);
            if (pq.size() > k) {
                pq.pop();          // 擠出前 k 名的,永遠回不來
            }
        }
    }

    int add(int val) {
        pq.push(val);              // 先 push:讓 val 和守門員公平比一次
        if (pq.size() > size) {
            pq.pop();              // 被丟的是 k+1 個裡最小的
        }
        return pq.top();
    }
};
