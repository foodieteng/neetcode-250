完整實作// MAX-HEAP GREEDY

p1405.cpp C++17 
class Solution {
public:
    string longestDiverseString(int a, int b, int c) {
        using P = pair<int, char>;
        priority_queue<P> pq;                               // 剩最多的在堆頂

        if (a) pq.push({a, 'a'});                           // 0 個不放
        if (b) pq.push({b, 'b'});
        if (c) pq.push({c, 'c'});

        string res = "";

        while (!pq.empty()) {
            auto [cnt1, ch1] = pq.top(); pq.pop();          // 最多的
            int n = res.size();

            if (n >= 2 && ch1 == res[n - 2] && res[n - 1] == ch1) {
                if (pq.empty()) return res;                 // 沒有別的字可插

                auto [cnt2, ch2] = pq.top(); pq.pop();      // 第二多的插一個
                res += ch2;
                if (--cnt2) pq.push({cnt2, ch2});
            }

            res += ch1;                                     // 再放最多的
            if (--cnt1) pq.push({cnt1, ch1});
        }

        return res;
    }
};
