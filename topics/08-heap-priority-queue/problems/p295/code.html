完整實作 · 我的版本// SIZE FIRST, THEN SWAP

p295.cpp C++17 
class MedianFinder {
public:
    priority_queue<int> mxH;                              // 較小的一半
    priority_queue<int, vector<int>, greater<int>> miH;   // 較大的一半

    MedianFinder() {
        
    }
    
    void addNum(int num) {
        if (mxH.size() <= miH.size()) {                   // 先顧數量
            mxH.push(num);
        } else {
            miH.push(num);
        }

        if (!miH.empty() && mxH.top() > miH.top()) {      // 再修順序
            auto a = miH.top(); miH.pop();
            auto b = mxH.top(); mxH.pop();
            mxH.push(a); miH.push(b);                     // size 不變
        }
    }
    
    double findMedian() {
        if (mxH.size() > miH.size()) {
            return (double)(mxH.top());
        }

        return double(mxH.top() + miH.top()) / 2.0;       // 先 int 相加
    }
};

完整實作 · 精簡版// ORDER FIRST, THEN BALANCE

p295_clean.cpp C++17 
class MedianFinder {
    priority_queue<int> lo;                               // 較小的一半
    priority_queue<int, vector<int>, greater<int>> hi;    // 較大的一半

public:
    void addNum(int num) {
        lo.push(num);                                     // ① 先進 lo
        hi.push(lo.top()); lo.pop();                      // ② lo 最大的送 hi → 順序
        if (hi.size() > lo.size()) {                      // ③ hi 多就送回 → 數量
            lo.push(hi.top()); hi.pop();
        }
    }

    double findMedian() {
        if (lo.size() > hi.size()) return lo.top();
        return (lo.top() + (double)hi.top()) / 2.0;       // 先轉 double
    }
};
我的版本精簡版
add 的步驟依 size 放 → 亂序就 swap進 lo → 送 hi → 多就送回
要比較 top 嗎要,還要先確認 !miH.empty()不用
每次 push / pop1 / 0,或 3 / 2(swap 時)固定 2 / 1,每兩次有一次(總數變奇數時)再 +1 / +1
5 萬次隨機實測99,632 / 49,632125,000 / 75,000
面試好講要解釋「為什麼交換一次就夠」兩行各管一條不變量
精簡版的每一行都只管一件事:② 保證順序(送過去的是 lo 最大的),③ 保證數量(hi 永遠不比 lo 多)。 我的版本操作比較少,精簡版比較難寫錯 —— 兩個都是 O(log n),選自己講得清楚的那個。
