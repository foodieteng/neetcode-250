完整實作// MIN-HEAP OF SIZE K

p215.cpp C++17 
class Solution {
public:
    int findKthLargest(vector<int>& nums, int k) {
        // 最小堆:top = 目前留下的 k 個裡最小的
        priority_queue<int, vector<int>, greater<int>> pq;
        for (int num : nums) {
            pq.push(num);                  // 重複值照樣進堆,不去重
            if (pq.size() > k) pq.pop();   // 踢掉 k+1 個裡最小的
        }

        return pq.top();                  // 最大的 k 個(含重複)裡最小的
    }
};
