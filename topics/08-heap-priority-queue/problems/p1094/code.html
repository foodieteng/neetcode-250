完整實作// SORT + MIN-HEAP

p1094.cpp C++17 
class Solution {
public:
    bool carPooling(vector<vector<int>>& trips, int capacity) {
        // [numPassengersi, fromi, toi]
        sort(trips.begin(), trips.end(), [](auto& t1, auto& t2) {
            return t1[1] < t2[1];                           // 依上車點排序
        });

        // {to, num}
        using P = pair<int, int>;
        priority_queue<P, vector<P>, greater<P>> pq;        // 最早下車的在堆頂
        int passenger = 0;

        for (auto& t : trips) {
            int num = t[0], from = t[1], to = t[2];

            while (!pq.empty() && pq.top().first <= from) { // 先下車(<=)
                passenger -= pq.top().second; pq.pop();
            }

            passenger += num;                               // 再上車
            if (passenger > capacity) return false;
            pq.emplace(to, num);
        }
        return true;
    }
};
