完整實作// MIN-HEAP OF SIZE 10 BY TIME

p355.cpp C++17 
class Twitter {
public:
    unordered_map<int, unordered_set<int>> followG;          // 誰追蹤誰
    unordered_map<int, vector<pair<int, int>>> tweets;    // user → (時間, tweetId)
    int ts = 0;                                            // 全域時間戳

    Twitter() {

    }

    void postTweet(int userId, int tweetId) {
        tweets[userId].push_back({++ts, tweetId});
    }

    vector<int> getNewsFeed(int userId) {
        auto cmp = [&](const auto& p1, const auto& p2) {
            return p1.first > p2.first;                    // 最小堆:top = 最舊
        };

        priority_queue<pair<int, int>, vector<pair<int, int>>, decltype(cmp)> pq(cmp);

        for (auto& [ts, tweetId] : tweets[userId]) {
            pq.push({ts, tweetId});
            if (pq.size() > 10) pq.pop();               // 踢最舊的
        }

        for (auto& followeeId : followG[userId]) {
            for (auto& [ts, tweetId] : tweets[followeeId]) {
                pq.push({ts, tweetId});
                if (pq.size() > 10) pq.pop();
            }
        }

        vector<int> ans;
        while (!pq.empty()) {
            ans.push_back(pq.top().second); pq.pop();   // 舊 → 新
        }
        reverse(ans.begin(), ans.end());                   // 新 → 舊
        return ans;
    }

    void follow(int followerId, int followeeId) {
        if (followerId == followeeId) return;              // 自己的推文已經加過
        followG[followerId].insert(followeeId);
    }

    void unfollow(int followerId, int followeeId) {
        if (followerId == followeeId) return;
        followG[followerId].erase(followeeId);
    }
};
