完整實作// ROUND SIMULATION

p621.cpp C++17 
class Solution {
public:
    int leastInterval(vector<char>& tasks, int n) {
        vector<int> freq(26, 0);
        for (char c : tasks) {
            freq[c - 'A']++;                 // 大寫!'a' 會算出 −32
        }

        priority_queue<int> pq;
        for (int f : freq) {
            if (f) pq.push(f);              // 跳過沒出現的任務
        }

        int cycle = n + 1;                     // 一輪 n+1 格
        int time = 0;
        vector<int> buff;                    // 這輪做過、正在冷卻的

        while (!pq.empty()) {
            buff.clear();
            for (int i = 0; i < cycle; i++) {
                if (pq.empty() && buff.empty()) break;   // 全部做完:不補 idle

                if (!pq.empty()) {
                    int cnt = pq.top(); pq.pop();   // 剩最多的先做
                    if (--cnt > 0) buff.push_back(cnt);
                }
                time++;                            // pq 空但 buff 有 → 這格是 idle
            }

            for (int f : buff) {                 // 輪結束,冷卻完畢
                pq.push(f);
            }
        }

        return time;
    }
};

🎤 面試模擬 · UMPIRE mock interview transcript ↗

time++ 放在 if (!pq.empty()) 外面,是這份程式最精巧的地方: 堆有東西 → 這格做任務;堆空了但 buff 有 → 這格是 idle,時間照樣要算。 idle 不用特別寫,它就是「這格沒 pop 到東西、但時間還是過了一格」。 -Wall -Wextra 編譯零警告。
