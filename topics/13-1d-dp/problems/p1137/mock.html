The table version first: guard the small cases, size the vector n + 1, set the three bases, then one loop that sums the previous three.

class Solution {
public:
    int tribonacci(int n) {
        if (n == 0) return 0;              // guard: small n has no room for t[2]
        if (n <= 2) return 1;

        vector<int> t(n + 1, 0);
        t[0] = 0; t[1] = t[2] = 1;        // three bases

        for (int i = 3; i <= n; i++)
            t[i] = t[i - 3] + t[i - 2] + t[i - 1]; // sum of the previous three

        return t[n];
    }
};

Then the O(1)-space version — only the last three values matter, so I keep a, b and c and roll them forward.

class Solution {
public:
    int tribonacci(int n) {
        if (n == 0) return 0;
        if (n <= 2) return 1;
        int a = 0, b = 1, c = 1;            // t[i-3], t[i-2], t[i-1]
        for (int i = 3; i <= n; i++) {
            int cur = a + b + c;            // t[i]
            a = b; b = c; c = cur;          // roll all three forward
        }
        return c;
    }
};

Both compute the same recurrence. One thing worth flagging — the shift a=b; b=c; c=cur; must run left to right, so each variable is read before it gets overwritten.
