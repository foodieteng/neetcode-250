Alright, let me code this up. In the table version, rob handles the tiny cases, then returns the max of take on the drop-tail range and take on the drop-head range.

class Solution {
public:
    int rob(vector<int>& nums) {
        int n = nums.size();
        if (n == 1) return nums[0];            // one house → no neighbor at all
        if (n == 2) return max(nums[0], nums[1]); // two houses are neighbors → take the larger

        // drop tail nums[0..n-2]  vs  drop head nums[1..n-1], take max
        return max(take(nums, 0, n - 2), take(nums, 1, n - 1));
    }

    // linear house robber, applied to the range nums[s..e]
    int take(vector<int>& nums, int s, int e) {
        int n = e - s + 1;
        vector<int> dp(n + 1, 0);
        dp[1] = nums[s];                     // first house of the range: just rob it

        for (int i = 2; i <= n; i++) {
            dp[i] = max(dp[i - 2] + nums[s + i - 1],  // rob
                        dp[i - 1]);                  // skip
        }
        return dp[n];
    }
};

take is the linear robber on a sub-range [s, e]: dp[i] is the best over the first i houses of that range, so the i-th house is nums[s + i - 1]. The O(1)-space take is the same recurrence with two rolling variables.

class Solution {
public:
    int rob(vector<int>& nums) {
        int n = nums.size();
        if (n == 1) return nums[0];
        return max(take(nums, 0, n - 2), take(nums, 1, n - 1));
    }

    int take(vector<int>& nums, int s, int e) {
        int prev2 = 0, prev1 = 0;        // dp[i-2], dp[i-1]
        for (int i = s; i <= e; i++) {
            int cur = max(prev1, prev2 + nums[i]); // skip vs rob
            prev2 = prev1;
            prev1 = cur;
        }
        return prev1;
    }
};

One thing worth flagging — compute cur first, then shift, so prev1 isn't clobbered before it feeds prev2. And the rolling version needs no guard for two houses: each range is a single house, and take simply returns it.
