int bfsGrid(vector<vector<int>>& g, int sr, int sc) {
    int m = g.size(), n = g[0].size();
    vector<vector<bool>> vis(m, vector<bool>(n));
    queue<pair<int,int>> q;
    q.push({sr, sc}); vis[sr][sc] = true;
    int steps = 0, d[4][2]={{1,0},{-1,0},{0,1},{0,-1}};
    while (!q.empty()) {
        int sz = q.size();            // 一次一整層
        for (int i = 0; i < sz; i++) {
            auto [r, c] = q.front(); q.pop();
            for (auto& e : d) {
                int nr=r+e[0], nc=c+e[1];
                if (nr<0||nr>=m||nc<0||nc>=n) continue;
                if (vis[nr][nc]||g[nr][nc]==0) continue;
                vis[nr][nc]=true; q.push({nr,nc});
            }
        }
        steps++;
    }
    return steps;
}
DIJKSTRA最短路 · 非負權O(E log V)
WHEN 非負權單源最短路
vector<long long> dijkstra(int s, int n,
        vector<vector<pair<int,int>>>& adj) {
    const long long INF = LLONG_MAX;
    vector<long long> dist(n, INF);
    priority_queue<pair<long long,int>,
        vector<pair<long long,int>>, greater<>> pq;
    dist[s] = 0; pq.push({0, s});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;   // 過期跳過
        for (auto& [v, w] : adj[u])
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
    }
    return dist;
}
DFS深度優先 · 遍歷 / 連通塊O(V+E)
WHEN 遍歷 · 路徑存在性 · 島嶼數
void dfs(int u, vector<vector<int>>& adj,
         vector<bool>& vis) {
    vis[u] = true;
    for (int v : adj[u])
        if (!vis[v]) dfs(v, adj, vis);
}

// 連通塊 / 島嶼數量
int numComponents(int n, vector<vector<int>>& adj) {
    vector<bool> vis(n, false);
    int cnt = 0;
    for (int i = 0; i < n; i++)
        if (!vis[i]) { dfs(i, adj, vis); cnt++; }
    return cnt;
}
BELLMAN最短路 · 負權 / 判負環O(V·E)
WHEN 有負權邊 · 偵測負環
vector<long long> bellmanFord(int s, int n,
        vector<array<int,3>>& edges) {   // {u,v,w}
    const long long INF = LLONG_MAX;
    vector<long long> dist(n, INF);
    dist[s] = 0;
    for (int i = 0; i < n - 1; i++)      // 鬆弛 n-1 輪
        for (auto& [u, v, w] : edges)
            if (dist[u] != INF && dist[u] + w < dist[v])
                dist[v] = dist[u] + w;
    for (auto& [u, v, w] : edges)        // 還能鬆弛 => 負環
        if (dist[u] != INF && dist[u] + w < dist[v])
            return {};
    return dist;
}
DSU並查集 · 壓縮＋按秩~O(α(n))
WHEN 動態合併群組 · 判連通 · 無向環
struct DSU {
    vector<int> p, rk;
    DSU(int n): p(n), rk(n, 0) { iota(p.begin(), p.end(), 0); }
    int find(int x) {
        while (p[x] != x) { p[x] = p[p[x]]; x = p[x]; }  // 壓縮
        return x;
    }
    bool unite(int a, int b) {
        int ra = find(a), rb = find(b);
        if (ra == rb) return false;      // 已同群
        if (rk[ra] < rk[rb]) swap(ra, rb);
        p[rb] = ra;
        if (rk[ra] == rk[rb]) rk[ra]++;
        return true;
    }
};
FLOYD全點對最短路 · n 小O(V³)
WHEN 所有點對最短路 (n≤400)
void floyd(vector<vector<long long>>& dist, int n) {
    // dist[i][j]=邊權, 無邊=INF, dist[i][i]=0
    const long long INF = LLONG_MAX;
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (dist[i][k]!=INF && dist[k][j]!=INF)
                    dist[i][j] = min(dist[i][j],
                        dist[i][k] + dist[k][j]);
}
TOPO拓撲排序 · Kahn (BFS)O(V+E)
WHEN 排課 / 依賴順序 · 順便判環
vector<int> topoSort(int n, vector<vector<int>>& adj) {
    vector<int> indeg(n, 0);
    for (int u = 0; u < n; u++)
        for (int v : adj[u]) indeg[v]++;
    queue<int> q;
    for (int i = 0; i < n; i++)
        if (indeg[i] == 0) q.push(i);
    vector<int> order;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : adj[u])
            if (--indeg[v] == 0) q.push(v);
    }
    // 長度 != n  =>  有環
    return (int)order.size() == n ? order : vector<int>{};
}
KRUSKAL最小生成樹 · 邊排序O(E log E)
WHEN MST（邊多）· 配 DSU
int kruskal(int n, vector<array<int,3>>& e) {  // {w,u,v}
    sort(e.begin(), e.end());
    DSU dsu(n);
    int cost = 0, used = 0;
    for (auto& [w, u, v] : e)
        if (dsu.unite(u, v)) { cost += w; used++; }
    return used == n - 1 ? cost : -1;   // -1=不連通
}
CYCLE環偵測 · 有向三色 DFSO(V+E)
WHEN 有向圖判環（灰色=環）
bool dfsTopo(int u, vector<vector<int>>& adj,
             vector<int>& st, vector<int>& order) {
    st[u] = 1;                 // 1=訪問中(灰)
    for (int v : adj[u]) {
        if (st[v] == 1) return false;      // 遇灰 => 環
        if (st[v] == 0 && !dfsTopo(v, adj, st, order))
            return false;
    }
    st[u] = 2;                 // 2=完成(黑)
    order.push_back(u);
    return true;
}
// 全跑完 reverse(order) 即拓撲序
PRIM最小生成樹 · 點擴張O(E log V)
WHEN MST（稠密）· 配 pq
int prim(int n, vector<vector<pair<int,int>>>& adj) {
    vector<bool> in(n, false);
    priority_queue<pair<int,int>,
        vector<pair<int,int>>, greater<>> pq;
    pq.push({0, 0});                // {權重, 起點}
    int cost = 0, cnt = 0;
    while (!pq.empty() && cnt < n) {
        auto [w, u] = pq.top(); pq.pop();
        if (in[u]) continue;
        in[u] = true; cost += w; cnt++;
        for (auto& [v, wt] : adj[u])
            if (!in[v]) pq.push({wt, v});
    }
    return cnt == n ? cost : -1;
}
BIPARTITE二分圖 · 兩色染色O(V+E)
WHEN 判二分圖 / 相鄰不同色
bool isBipartite(int n, vector<vector<int>>& adj) {
    vector<int> color(n, -1);
    for (int s = 0; s < n; s++) {
        if (color[s] != -1) continue;
        queue<int> q; q.push(s); color[s] = 0;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : adj[u]) {
                if (color[v] == -1) {
                    color[v] = color[u] ^ 1; q.push(v);
                } else if (color[v] == color[u])
                    return false;      // 同色相鄰
            }
        }
    }
    return true;
}

背誦口訣
  • 無權最短路 → BFS 分層;非負權 → Dijkstra + 小根堆;負權 → Bellman-Ford 鬆弛 n-1 輪
  • 連通 / 合併 → DSU（find 壓縮、unite 按秩）
  • 排順序 / 判依賴環 → 拓撲（Kahn 數入度,長度 ≠ n 即有環）
  • 最小生成樹 → 邊多 Kruskal、稠密 Prim,皆貪心
  • 易錯：Dijkstra 記得 greater<> 小根堆 + 過期即跳 d > dist[u];帶權一律用 long long 防溢位
