class Solution {
    static const int MAXV = 100001;   // 值域上限 +1（nums[i] ≤ 1e5）
    vector<int> parent, spf;       // parent:DSU;spf:最小質因數

    int Find(int x) {
        return parent[x] < 0 ? x : parent[x] = Find(parent[x]);  // 路徑壓縮
    }

    void Union(int x, int y) {
        int rx = Find(x), ry = Find(y);
        if (rx == ry) return;
        if (-parent[rx] < -parent[ry]) swap(rx, ry);  // rx = 較大分量,較小者掛上去
        parent[rx] += parent[ry];          // 負值 = 分量大小,累加
        parent[ry] = rx;
    }

public:
    bool canTraverseAllPairs(vector<int>& nums) {
        if (nums.size() == 1) return true;   // 單一元素:沒有任何「一對」要連

        // ── 線性篩:一次算出每個數的最小質因數 spf ──
        spf.assign(MAXV, 0);              // spf[i]=0 代表「還沒被標記」
        for (int i = 2; i < MAXV; i++)
            if (!spf[i])                    // i 從沒被標記 ⇒ i 是質數
                for (int j = i; j < MAXV; j += i)
                    if (!spf[j]) spf[j] = i;   // 先到先得:最小質數先蓋

        parent.assign(MAXV, -1);          // DSU 的節點 = 質數值本身

        // ── 把每個數的所有質因數併成一組;相鄰數靠共用質數串起 ──
        for (int x : nums) {
            if (x == 1) return false;       // 含 1 且 n>1 ⇒ 1 與誰都 gcd=1
            int first = spf[x];              // 用 x 的最小質因數當錨點
            while (x > 1) {
                int p = spf[x];            // 當前最小質因數
                Union(first, p);            // first 和 p 同屬 x → 併起來
                while (x % p == 0) x /= p;  // 把這個質因數剝光
            }
        }

        // ── 連通性檢查:用「質因數」的根,不是合成數 ──
        int root = Find(spf[nums[0]]);
        for (int x : nums)
            if (Find(spf[x]) != root) return false;
        return true;
    }
};

複雜度分析// COMPLEXITY

步驟成本說明
assign × 2O(A)parent、spf 兩個長度 MAXV 的陣列
篩 spf[]O(A log log A)埃氏篩,只寫未被標記者(!spf[j])
拆質因數 + unionO(n log A · α)每數 ≤ log₂A 個質因數
連通性檢查O(n · α)每數一次 Find
T = O(A log log A + n log A)　　S = O(A)

A = 值域上限 MAXV ≈ 10⁵（篩固定跑滿整個值域),n = 陣列長度。α 為反 Ackermann,視為常數。

維度量級
TimeO(A log log A + n log A)
SpaceO(A)

邊界與陷阱// PITFALLS

陷阱後果正解
篩的守門忘了 !spf[j] 存到最大質因數 / 篩失效 只在 !spf[j](還沒被標記)時寫入,才拿到最小質因數
檢查用合成數 Find(x) 越界 / 錯根 union 的是質數,檢查也要用質因數 Find(spf[x])
單一元素 n == 1 誤判 false 先行 return true(沒有任何「一對」需要連)
陣列含 1 且 n > 1 1 與任何數 gcd=1 迴圈遇到 x == 1 立即 return false
parent/spf 開得不夠大 以質數值當索引會越界 固定開到 MAXV = 100001(> 值域上限,免算 max)

細節走讀// WALKTHROUGH

SIEVE 原理 · 最小質因數篩(smallest prime factor)

要解決什麼:本題要對每個數做質因數分解。若對每個數 trial division 試到 √x, n 個數就是 O(n·√A)。改成預先算一張表 spf[x] = x 的最小質因數, 之後任何數都能靠這張表在 O(log x) 內拆完。

篩怎麼建(埃氏篩變體):外層 i 從 2 遞增。spf[i]==0 代表 i 從沒被任何更小的數標記過 —— 這正是「i 是質數」的定義。於是用 i 去標記它的所有倍數 j = i, 2i, 3i…, 但只在 spf[j] 還是 0 時才寫 spf[j] = i。

為什麼寫進去的一定是「最小」質因數:因為外層 i 由小到大, 第一個標到 j 的質數,必定是 j 的最小質因數 —— 之後更大的質數再遇到 j 時 spf[j] 已非 0, 守門 !spf[j] 擋下,不再覆蓋。這就是「先到先得」。整體 O(A log log A)(質數調和級數)。

// SIEVE: 建表小範例(看 spf 怎麼被填) i=2 質數 → 標 4,6,8,10,12…:spf[4]=spf[6]=spf[8]=spf[10]=spf[12]=2 i=3 質數 → 標 9,15,21…（6,12 已是 2,守門 !spf 擋下不覆蓋）:spf[9]=spf[15]=3 i=4 → spf[4]=2≠0,非質數,跳過(不進內層) i=5 質數 → spf[10] 已是 2 擋下;spf[25]=5 … ↳ 結果:每個合成數存的都是它「最小」的質因數
// FACTORIZE: while (x > 1) 用 spf 逐層剝質因數 first = spf[x] 取 x 的最小質因數當「錨」 迴圈:p = spf[x]; Union(first, p); 再 while(x%p==0) x/=p 把 p 剝光 ↳ x 的每個相異質因數都和 first 併在一起 ⇒ x 的質因數同一組 例:x=12 → first=spf[12]=2 p=2 Union(2,2)無效; 剝成 x=3 p=spf[3]=3 Union(2,3); 剝成 x=1 停 → {2,3} 同組
// UNION: union by size + 路徑壓縮 parent[root] 存負的分量大小;-parent 越大代表分量越大 合併時讓較大分量當新根(rx),較小者掛上去 → 樹更矮 Find 回傳過程中把節點直接指向根(路徑壓縮)→ 攤還近 O(1)
