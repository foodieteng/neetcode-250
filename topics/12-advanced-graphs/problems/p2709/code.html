完整實作 · 質因數併查集// SPF + DSU

class Solution {
    vector<int> parent;                // 併查集:負值 = 分量大小（union by size）

    int Find(int x) {
        return parent[x] < 0 ? x : parent[x] = Find(parent[x]);  // 路徑壓縮
    }

    bool Union(int x, int y) {
        int rootX = Find(x), rootY = Find(y);
        if (rootX == rootY) return false;
        if (-parent[rootX] < -parent[rootY]) swap(rootX, rootY);  // rootX = 較大分量
        parent[rootX] += parent[rootY];
        parent[rootY] = rootX;
        return true;
    }

public:
    bool canTraverseAllPairs(vector<int>& nums) {
        int n = nums.size();
        if (n == 1) return true;             // 單一元素:沒有任何「一對」要連

        int mx = *max_element(nums.begin(), nums.end());
        if (mx == 1) return false;           // n>1 卻全是 1 ⇒ 兩兩 gcd=1

        parent.assign(mx + 1, -1);
        vector<int> spf(mx + 1);
        iota(spf.begin(), spf.end(), 0);      // spf[i] 初值 = i（先假設自己是最小質因數）

        // ── 篩最小質因數 spf ──  ← bug ①:守門必須是 spf[j]==j
        for (int i = 2; i * i <= mx; i++) {
            if (spf[i] == i) {                // spf[i]==i ⇒ i 沒被更小質數蓋過 ⇒ i 是質數
                for (int j = i * i; j <= mx; j += i)
                    if (spf[j] == j) spf[j] = i;  // 先到先得 = 最小質因數
            }
        }

        // ── 把每個數的所有質因數併成一組 ──
        for (int v : nums) {
            if (v == 1) return false;         // 含 1 且 n>1 ⇒ 走不到
            int first = spf[v];              // 用最小質因數當錨
            while (v > 1) {
                int p = spf[v];
                Union(p, first);            // v 的每個質因數都併到 first
                while (v % p == 0) v /= p;  // 剝光這個質因數
            }
        }

        // ── 連通性檢查 ──  ← bug ②:要用質因數 spf[v],不是合成數 v
        int anchor = spf[nums[0]];
        for (int v : nums)
            if (Find(anchor) != Find(spf[v])) return false;
        return true;
    }
};

TWO BUGS · 這題最容易死在這兩行

① 篩的守門是 spf[j] == j,不是 == i。 spf 初值被 iota 設成 spf[j]=j。守門要問的是「j 還沒被更小質數蓋過嗎」,那就是 spf[j]==j。 若誤寫 spf[j]==i:spf[j] 初值是 j,幾乎永遠不等於當前質數 i,條件不成立 ⇒ 篩形同虛設,spf 全留 j,之後分解合成數整個崩掉。

② 連通檢查用 Find(spf[v]),不是 Find(v)。 我們 Union 的節點是質數,合成數 v(如 12)從沒被 union 過。Find(v) 會回到它自己、與任何質數不同根,直接誤判 false。 取 spf[v](v 的一個質因數,必在 v 的分量裡)才代表得了 v。

A = 陣列最大值 mx(≤ 10⁵),n = 陣列長度。α 為反 Ackermann,視為常數。

TimeO(A log log A + n log A)

細節走讀// WALKTHROUGH

SIEVE 原理 · 為什麼守門是 spf[j] == j

要解決什麼:本題要對每個數做質因數分解。與其每個數 trial division 試到 √v,不如預先算一張表 spf[x] = x 的最小質因數,之後任何數靠這表在 O(log x) 內拆完。

怎麼建:iota 先把 spf[i] = i(每個數暫時假設自己是最小質因數)。外層 i 由小到大:spf[i] == i 代表 i 從沒被任何更小質數改過 —— 這正是「i 是質數」。於是用 i 去標記倍數 j,但只在 spf[j] == j(j 還沒被蓋過)時才寫 spf[j] = i。

為什麼寫進去的一定是「最小」:外層 i 遞增,第一個標到 j 的質數必是 j 的最小質因數;之後更大的質數再遇到 j 時 spf[j]≠j,守門擋下不覆蓋 —— 這就是「先到先得」。整體 O(A log log A)。

// SIEVE: 建表小範例(看 spf 怎麼被填) iota 後:spf[i] = i（4→4, 6→6, 9→9 …） i=2 質數(spf[2]==2)→ 標 j=4,6,8,10,12…:凡 spf[j]==j 者寫 2 ↳ spf[4]=spf[6]=spf[8]=spf[10]=spf[12]=2 i=3 質數(spf[3]==3)→ j=9,15,21…；但 6,12 已是 2(spf[j]≠j)守門擋下 ↳ spf[9]=spf[15]=3 ↳ 結果:每個合成數存的都是它「最小」的質因數（先到先得）
// FACTORIZE: first = spf[v] 當錨,逐層剝質因數 first = spf[v] 取 v 的最小質因數當「錨」 迴圈:p = spf[v]; Union(p, first); 再 while(v%p==0) v/=p 把 p 剝光 ↳ v 的每個相異質因數都和 first 併在一起 ⇒ v 的質因數同一組 例:v=12 → first=spf[12]=2 p=2 Union(2,2) 無效; 剝成 v=3 p=spf[3]=3 Union(3,2); 剝成 v=1 停 → {2,3} 同組
// CHECK: anchor = spf[nums[0]],逐一比對根 anchor = spf[nums[0]] 取第一個數的質因數當基準 每個 v:Find(spf[v]) 是否 == Find(anchor)？全同 ⇒ 全連通 ↳ 用 spf[v]（質因數）不是 v（合成數）—— 這就是 bug ②
