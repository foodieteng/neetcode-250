class Solution {
private:
    vector<int> parent;

    int Find(int x) {
        return parent[x] < 0 ? x : parent[x] = Find(parent[x]);
    }

    void Union(int x, int y) {
        int a = Find(x), b = Find(y);
        if (a == b) return;
        if (-parent[a] < -parent[b]) swap(a, b);
        parent[a] += parent[b];
        parent[b] = a;
    }

public:
    int swimInWater(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        auto getId = [&](int r, int c) { return r * n + c; };
        parent.assign(m * n, -1);

        const vector<pair<int, int>> dirs = {{-1,0},{1,0},{0,-1},{0,1}};
        vector<bool> on(m * n, false);

        using P = pair<int, int>;
        priority_queue<P, vector<P>, greater<P>> pq;   // (height, id)
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                pq.push({grid[i][j], getId(i, j)});
            }
        }

        while (!pq.empty()) {
            auto [h, id] = pq.top();
            pq.pop();
            on[id] = true;                    // flood this cell
            int r = id / n, c = id % n;

            for (auto [dr, dc] : dirs) {
                int nr = r + dr, nc = c + dc;
                if (nr < 0 || nc < 0 || nr >= m || nc >= n) continue;
                int nid = getId(nr, nc);
                if (on[nid]) Union(id, nid);   // join flooded neighbor
            }

            if (Find(0) == Find(getId(m - 1, n - 1))) {
                return h;                    // start meets end → answer
            }
        }
        return -1;
    }
};

One thing worth flagging is the connectivity check right after each flood: Find(0) == Find(getId(m - 1, n - 1)). The heights come out of the min-heap in increasing order, so the first time this passes, h is the minimum level that links the corners.

Now the same minimax two other ways. Dijkstra, but I relax with max, not +: a neighbor's cost is the max of my cost and its height.

class Solution {
public:
    int swimInWater(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        using T = tuple<int, int, int>;            // (bottleneck, r, c)
        priority_queue<T, vector<T>, greater<T>> pq;

        const vector<pair<int, int>> dirs = {{-1,0},{1,0},{0,-1},{0,1}};
        vector<vector<int>> dist(m, vector<int>(n, -1));

        dist[0][0] = grid[0][0];
        pq.emplace(grid[0][0], 0, 0);

        while (!pq.empty()) {
            auto [h, r, c] = pq.top();
            pq.pop();

            if (dist[r][c] != -1 && h > dist[r][c]) continue;   // stale entry
            if (r == m - 1 && c == n - 1) return h;

            for (auto [dr, dc] : dirs) {
                int nr = r + dr, nc = c + dc;
                if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue;

                int d = max(h, grid[nr][nc]);          // cost = tallest cell so far
                if (dist[nr][nc] == -1 || dist[nr][nc] > d) {
                    dist[nr][nc] = d;
                    pq.emplace(d, nr, nc);
                }
            }
        }
        return -1;
    }
};

And binary search on the answer, since a bigger t only ever floods more cells; a BFS checks reachability.

class Solution {
public:
    int swimInWater(vector<vector<int>>& grid) {
        int n = grid.size();
        int l = max(grid[0][0], grid[n - 1][n - 1]);   // lower bound: start/end heights
        int r = n * n - 1;                              // upper bound: max height

        while (l < r) {
            int mid = l + (r - l) / 2;
            if (bfs(grid, mid)) {
                r = mid;                                 // reachable → shrink the upper bound
            } else {
                l = mid + 1;                             // unreachable → raise the lower bound
            }
        }
        return l;
    }

private:
    bool bfs(const vector<vector<int>>& grid, int t) {
        int n = grid.size();
        if (grid[0][0] > t) return false;

        queue<pair<int, int>> q;
        vector<vector<bool>> seen(n, vector<bool>(n, false));
        q.push({0, 0});
        seen[0][0] = true;

        const vector<pair<int, int>> dirs = {{-1,0},{1,0},{0,-1},{0,1}};

        while (!q.empty()) {
            auto [r, c] = q.front();
            q.pop();
            if (r == n - 1 && c == n - 1) return true;

            for (const auto& [dr, dc] : dirs) {
                int nr = r + dr, nc = c + dc;
                if (nr < 0 || nr >= n || nc < 0 || nc >= n) continue;
                if (seen[nr][nc] || grid[nr][nc] > t) continue;
                seen[nr][nc] = true;
                q.push({nr, nc});
            }
        }
        return false;
    }
};

Across the three: union-find relies on adding cells in sorted order, Dijkstra on max being monotonic so the greedy pop is safe, and binary search on reachability being monotonic in t. Same minimax fact, three angles.

Let's dry-run the waterline on that grid, [[0,2,1],[5,8,3],[6,7,4]] — expected 4. I flood cells in height order and union with flooded neighbors.

hflood cellcomponents
0(0,0) start{start}
1(0,2){start}, {(0,2)}
2(0,1)merges → top row joined
3(1,2)joins (0,2)
4(2,2) endjoins (1,2) → start == end → 4

Heights 0 to 3 build up the start, the top row, and the cell above the corner into one component. The row that matters is height 4: flooding the corner links it to that already-connected low path, so start and end share a component and I return 4. Heights 5 through 8 are never flooded. That matches.

Corner cases:

• A 1×1 grid returns its only height, grid[0][0].
• The answer is always at least max(grid[0][0], grid[n-1][n-1]), since you must stand on both — that's exactly the binary-search lower bound.

Let me think about complexity here.

All three are O(n² log n). Union-find and Dijkstra pay the log for the heap over n² cells; binary search does O(log n²) rounds of an O(n²) BFS.

The space complexity is O(n²), for the parent array, the dist grid, or the seen grid.

In the worst-case scenario the whole grid is one giant blob, but each of the three still stays within that bound.

Oh, one more thing — if I had to pick one, I'd take Dijkstra: it only explores cells reachable from the start and stops the instant it pops the corner, whereas the union-find version pushes all n² cells into the heap up front and floods every low cell, connected or not.
