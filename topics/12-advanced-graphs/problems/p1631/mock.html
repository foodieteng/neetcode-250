class Solution {
private:
    vector<int> parent;

    int Find(int x) {
        return parent[x] < 0 ? x : parent[x] = Find(parent[x]);
    }

    void Union(int x, int y) {
        int a = Find(x), b = Find(y);
        if (a == b) return;
        if (-parent[a] < -parent[b]) swap(a, b);
        parent[a] += parent[b];
        parent[b] = a;
    }

public:
    int minimumEffortPath(vector<vector<int>>& heights) {
        int m = heights.size(), n = heights[0].size();
        auto id = [&](int r, int c) { return r * n + c; };

        vector<tuple<int, int, int>> edges;        // (|Δ|, u, v)
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (j + 1 < n) {
                    int w = abs(heights[i][j] - heights[i][j + 1]);
                    edges.push_back({w, id(i, j), id(i, j + 1)});
                }
                if (i + 1 < m) {
                    int w = abs(heights[i][j] - heights[i + 1][j]);
                    edges.push_back({w, id(i, j), id(i + 1, j)});
                }
            }
        }
        sort(edges.begin(), edges.end());            // tuples sort by their first field, |Δ|

        parent.assign(m * n, -1);
        int src = id(0, 0), dst = id(m - 1, n - 1);
        for (auto& [w, u, v] : edges) {
            Union(u, v);
            if (Find(src) == Find(dst)) return w;     // connected → current weight = answer
        }
        return 0;
    }
};

One thing worth flagging: the tuples are (weight, u, v), so a plain sort already orders them by weight — no custom comparator needed. And the return w fires on the first edge that connects the corners.

Now the same minimax two other ways. Dijkstra, relaxing with max: the new effort is the max of the current effort and the next step, and popping the corner gives the answer.

class Solution {
public:
    int minimumEffortPath(vector<vector<int>>& heights) {
        int m = heights.size(), n = heights[0].size();
        vector<vector<int>> dist(m, vector<int>(n, INT_MAX));
        dist[0][0] = 0;

        const vector<pair<int, int>> dirs = {{-1,0},{1,0},{0,-1},{0,1}};
        using T = tuple<int, int, int>;               // (effort, r, c)
        priority_queue<T, vector<T>, greater<T>> pq;
        pq.push({0, 0, 0});

        while (!pq.empty()) {
            auto [d, r, c] = pq.top();
            pq.pop();
            if (r == m - 1 && c == n - 1) return d;    // popping the target = answer
            if (d > dist[r][c]) continue;              // stale entry

            for (auto [dr, dc] : dirs) {
                int nr = r + dr, nc = c + dc;
                if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue;
                int nd = max(d, abs(heights[nr][nc] - heights[r][c]));
                if (nd < dist[nr][nc]) {            // effort = max |Δ| along the path
                    dist[nr][nc] = nd;
                    pq.push({nd, nr, nc});
                }
            }
        }
        return 0;
    }
};

And binary search on the effort limit, since a bigger limit only ever unlocks more edges; a BFS checks whether the corner is reachable.

class Solution {
public:
    int minimumEffortPath(vector<vector<int>>& heights) {
        int R = heights.size(), C = heights[0].size();
        int lo = 0, hi = 0;
        for (int i = 0; i < R; i++) {
            for (int j = 0; j < C; j++) {
                if (j + 1 < C) hi = max(hi, abs(heights[i][j + 1] - heights[i][j]));
                if (i + 1 < R) hi = max(hi, abs(heights[i + 1][j] - heights[i][j]));
            }
        }

        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (reach(heights, mid)) hi = mid;       // reachable → shrink the upper bound
            else lo = mid + 1;                       // unreachable → raise the lower bound
        }
        return lo;
    }

private:
    bool reach(const vector<vector<int>>& h, int limit) {
        int R = h.size(), C = h[0].size();
        queue<pair<int, int>> q;
        vector<vector<bool>> seen(R, vector<bool>(C, false));
        q.push({0, 0});
        seen[0][0] = true;

        const vector<pair<int, int>> dirs = {{-1,0},{1,0},{0,-1},{0,1}};
        while (!q.empty()) {
            auto [r, c] = q.front();
            q.pop();
            if (r == R - 1 && c == C - 1) return true;

            for (auto [dr, dc] : dirs) {
                int nr = r + dr, nc = c + dc;
                if (nr < 0 || nr >= R || nc < 0 || nc >= C || seen[nr][nc]) continue;
                if (abs(h[nr][nc] - h[r][c]) > limit) continue;   // edge over the limit, can't cross
                seen[nr][nc] = true;
                q.push({nr, nc});
            }
        }
        return false;
    }
};

Across the three: union-find relies on sorted order, Dijkstra on max being monotonic so the greedy pop is safe, and binary search on reachability being monotonic in the limit. Same minimax fact, three angles.

Let's dry-run union-find on that grid, [[1,2,2],[3,8,2],[5,3,5]] — expected 2. I add edges by |diff| and union.

|diff|edges addedcomponents
0(0,1)-(0,2), (0,2)-(1,2)one island forms
1(0,0)-(0,1)src joins that island
2(0,0)-(1,0) … (2,1)-(2,2)dst joins → src == dst → 2

The zero edges link the top-right cells into one island, and the one edge pulls the start into it. The row that matters is weight 2: walking the weight-2 edges down the left and across the bottom finally pulls the corner in, so I return 2. The steep middle 8, with edges of 5 and 6, is never added. That matches.

Corner cases:

• A 1×1 grid has no edges, so effort is 0.
• A flat grid where all neighbors are equal is also 0 — the loop connects them at weight 0 immediately.

Let me think about complexity here.

All three are O(mn · log). Union-find sorts about 2mn edges; Dijkstra pays the log on the heap; binary search does O(log maxDiff) rounds of an O(mn) BFS over the grid.

The space complexity is O(mn), for the parent array, the dist grid, or the seen grid.

In the worst-case scenario every cell is explored, but each of the three stays within that bound.

Oh, one more thing — if the caller also wants the actual path, I'd take the Dijkstra version and store a parent pointer on each successful relax, then walk it back from the corner. The other two only give the effort value.
