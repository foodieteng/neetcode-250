Tarjan 求橋// dfn / low + parent 節點

p1192.cpp C++17 
class Solution {
public:
    vector<vector<int>> graph;
    vector<int> dfn, low;          // dfn=發現時間;low=能回到的最早 dfn
    int timer = 0;

    vector<vector<int>> criticalConnections(int n, vector<vector<int>>& connections) {
        graph.assign(n, {});
        dfn.assign(n, -1);              // -1 = 尚未訪問
        low.assign(n, -1);

        for (auto& e : connections) {
            graph[e[0]].push_back(e[1]);
            graph[e[1]].push_back(e[0]);
        }

        vector<vector<int>> ans;
        for (int i = 0; i < n; i++)
            if (dfn[i] == -1) dfs(-1, i, ans);  // 每個未訪連通塊都要起 DFS

        return ans;
    }

    void dfs(int p, int u, vector<vector<int>>& ans) {
        dfn[u] = low[u] = timer++;       // 發現 u:時間戳,low 初值 = 自己
        for (int v : graph[u]) {
            if (v == p) continue;            // 不走回父親(LC 1192 無重邊,安全)
            if (dfn[v] == -1) {            // 樹邊:v 是新節點
                dfs(u, v, ans);
                low[u] = min(low[u], low[v]);  // 繼承子樹能回到的最早
                if (low[v] > dfn[u]) ans.push_back({u, v});  // 逃不出 → 橋!
            } else {                        // 回邊:v 是已訪祖先
                low[u] = min(low[u], dfn[v]);  // 用 dfn[v],不是 low[v]
            }
        }
    }
};

前提 · 為什麼這裡能用 v == p(以及它的極限)

if (v == p) continue; 是用父節點擋回頭。它只在「u 和父親之間最多一條邊」時正確 —— 而 LC 1192 保證「沒有重複連線」,所以這版安全、也最乾淨。

它的極限:若允許重邊(u—v 有兩條平行邊),在 v 看到第二條回 u 的邊時,v == p 會把它一起 skip 掉 —— 但那其實是一條真正的備援回邊,漏掉它 low[v] 壓不下來,(u,v) 就被誤判成橋。通用寫法要改成傳「進來的邊 id」pe,只 skip 走過來那一條(見下方走讀)。
