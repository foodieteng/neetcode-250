class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        stack<int> st;
        string op = "+-*/";

        for (string token : tokens) {
            if (op.find(token) == string::npos) {  // 不是運算子 → 數字
                st.push(stoi(token));
                continue;
            }

            int y = st.top(); st.pop();          // 先彈 = 右運算元
            int x = st.top(); st.pop();          // 後彈 = 左運算元
            if (token == "+") {
                st.push(x + y);
            } else if (token == "-") {
                st.push(x - y);              // x - y,別顛倒
            } else if (token == "*") {
                st.push(x * y);
            } else {
                st.push(x / y);              // x / y,向零截斷
            }
        }
        return st.top();
    }
};

🎤 面試模擬 · UMPIRE mock interview transcript ↗

op.find(token) == npos 用來判「是不是數字」:運算子都是單一字元,數字(含多位或負號如 "-11")不會是 "+-*/" 的子字串,所以 find 回 npos → 當數字。 關鍵兩行是 y 先彈、x 後彈,再算 x op y。

複雜度分析// COMPLEXITY

步驟成本說明
掃 tokenO(n)每個 token 處理一次
push / pop / 運算O(1)每次常數
堆疊大小O(n)最多裝 ~n/2+1 個數
T = O(n)　　S = O(n)

n = token 數。一趟線性掃,無巢狀。

邊界與陷阱// PITFALLS

陷阱後果正解
x、y 順序顛倒 -、/ 算錯(y-x) 先彈 y(右)、後彈 x(左),算 x op y
判運算子時漏了負數 token 把 "-3" 誤當運算子 op.find(token)==npos 對 "-3" 回 npos → 數字 ✓
用 token[0] 判斷 "-3" 首字是 '-' → 誤判 比對整個 token 字串,或先判長度
忘了 stoi 轉型 push 到 string / 型別錯 數字 token 要 stoi 成 int 再 push
擔心負數除法 C++ / 本就向零截斷 與題目要求一致,直接用 x / y

細節走讀// WALKTHROUGH

// ["2","1","+","3","*"] 逐步(堆疊右=top) "2": 數字 → push 2 堆疊 [2] "1": 數字 → push 1 堆疊 [2, 1] "+": y=pop()=1, x=pop()=2 push 2+1=3 堆疊 [3] "3": 數字 → push 3 堆疊 [3, 3] "*": y=pop()=3, x=pop()=3 push 3*3=9 堆疊 [9] // return st.top() = 9
// 為什麼順序重要:["3","-4","+"] 與減法 "3" → push 3;"-4" → op.find("-4")==npos → 數字 → push -4 "+" → y=-4, x=3 → 3 + (-4) = -1 (加法交換律,順序無妨) 但若是 "-":x - y = 3 - (-4) = 7,絕不能寫成 y - x
"-4" 這種負數 token 是最容易踩的坑:因為它含 '-'。這裡靠 op.find(token)==npos(比對整個字串)正確判成數字 —— 若改用「首字是不是 +-*/」就會誤判。
