class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        stack<int> st;

        for (int ast : asteroids) {
            if (st.empty()) { st.push(ast); continue; }

            bool survived = true;

            while (!st.empty() && st.top() > 0 && ast < 0) {   // 左正 + 右負 才會撞
                if (st.top() == abs(ast)) {            // ① 相等 → 兩顆都炸
                    st.pop();
                    survived = false;
                    break;
                }

                if (st.top() > abs(ast)) {            // ② top 較大 → ast 炸
                    survived = false;
                    break;
                } else {                              // ③ top 較小 → top 炸,ast 續撞
                    st.pop();
                }
            }

            if (survived) st.push(ast);            // 沒被判爆 → 存活入堆疊
        }

        vector<int> ans;
        while (!st.empty()) { ans.push_back(st.top()); st.pop(); }
        reverse(ans.begin(), ans.end());       // stack 倒出來要反轉回原序
        return ans;
    }
};

🎤 面試模擬 · UMPIRE mock interview transcript ↗

while 的守門 st.top() > 0 && ast < 0 是整個解的核心:只有這個方向組合會相撞。 內部三個分支對應三種結果,其中只有「top < |ast| → pop top」會不 break、讓 ast 繼續往下連環撞。
2

複雜度分析// COMPLEXITY

觀察成本說明
每顆 push ≤ 1 次O(n)迴圈外層一趟
每顆 pop ≤ 1 次O(n)while 每轉一圈 pop 掉一顆(永久)
倒出 + reverseO(n)組答案
T = O(n) 攤還　　S = O(n)

while 看似巢狀,但總 pop 次數上限是 n(每顆只能被炸一次),所以整體仍線性。

3

邊界與陷阱// PITFALLS

陷阱後果正解
while 守門漏了方向 兩正/兩負/負左正右也去撞 必須 top > 0 && ast < 0
相等時忘了 pop top 兩顆都該炸卻留下 top == 分支要 pop() + survived=false
「top 較小」分支忘了不 break 連環撞斷掉,少炸幾顆 只 pop、不 break,讓 ast 續撞
用 survived 後仍無腦 push 已爆的 ast 又被放回 只有 survived 為真才 push
忘了最後 reverse 答案順序顛倒 stack 倒出是反序 → reverse
