解法 A · Trie + 由右往左// +1 · WALK · BREAK ON MISSING CHILD

p2707_trie.cpp C++17 
class TrieNode {
public:
    TrieNode *child[26] = { nullptr };
    bool isWord = false;
};

class Solution {
public:
    TrieNode *root;

    int minExtraChar(string s, vector<string>& dictionary) {
        root = new TrieNode();

        for (auto& w : dictionary) insert(w);

        int n = s.length();
        vector<int> dp(n + 1, 0);                      // dp[n] = 0:空後綴

        for (int start = n - 1; start >= 0; start--) {
            dp[start] = dp[start + 1] + 1;             // ① 先當 s[start] 是多的
            TrieNode* node = root;                     // ② 每個起點都從 root 走
            for (int end = start; end < n; end++) {
                int idx = s[end] - 'a';
                if (!node->child[idx]) break;          // ③ 路斷了:不會有更長的字
                node = node->child[idx];
                if (node->isWord) dp[start] = min(dp[start], dp[end + 1]);  // ④ 不 break
            }
        }

        return dp[0];
    }

    void insert(string& w) {
        TrieNode *cur = root;

        for (char& c : w) {
            int i = c - 'a';
            if (!cur->child[i]) cur->child[i] = new TrieNode();
            cur = cur->child[i];
        }
        cur->isWord = true;
    }
};

解法 B · unordered_set + 由左往右// +1 · TRY EVERY WORD ENDING AT i

p2707_set.cpp C++17 
class Solution {
public:
    int minExtraChar(string s, vector<string>& dictionary) {
        int n = s.size();
        vector<int> dp(n + 1, 0);                      // dp[0] = 0:空前綴
        unordered_set<string> dict(dictionary.begin(), dictionary.end());

        for (int i = 1; i <= n; i++) {
            dp[i] = dp[i - 1] + 1;                     // ① 先當 s[i-1] 是多的
            for (string w : dict) {                    // ② 每個字都比一次
                int L = w.size();
                if (i - L >= 0 && (s.substr(i - L, L) == w)) {  // ③ 結尾在 i 的那段
                    dp[i] = min(dp[i], dp[i - w.size()]);       // ④ 不 break
                }
            }
        }
        return dp[n];
    }
};
兩種寫法:官方範例 1 → 1、範例 2 → 3,都正確;和記憶化遞迴的參考解隨機對拍 40,000 組,0 筆不一致。 對拍輸入:字母 a–c,九成是 |s| ≤ 20、字典 ≤ 8 個短字;一成是 |s| = 50、字典 1–50 個字(含長到 50 的字)。

B 的兩個小問題,都和正確性無關,只影響速度:

  • unordered_set 只被走訪、從沒被查找,雜湊完全沒用上;直接掃 dictionary 結果一樣(題目保證字不重複)。
  • for (string w : dict) 每圈複製一次字;s.substr(i − L, L) 每次比對又配置一個新字串。改成 const string& 和 s.compare(i − L, L, w) == 0 就都不用配置。
p2707_set_fast.cpp C++17 
class Solution {
public:
    int minExtraChar(string s, vector<string>& dictionary) {
        int n = s.size();
        vector<int> dp(n + 1, 0);

        for (int i = 1; i <= n; i++) {
            dp[i] = dp[i - 1] + 1;
            for (const string& w : dictionary) {       // 不複製字
                int L = w.size();
                if (i - L >= 0 && s.compare(i - L, L, w) == 0)  // 不配置新字串
                    dp[i] = min(dp[i], dp[i - L]);
            }
        }
        return dp[n];
    }
};

改良版和 A、B 對拍 20,000 組 0 錯。速度見第 3 節。
