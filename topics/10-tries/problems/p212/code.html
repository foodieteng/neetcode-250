class TrieNode {
public:
    TrieNode *child[26] = { nullptr };
    string word = "";                     // 結尾節點存整個字
};

class Solution {
public:
    TrieNode *root;

    const vector<pair<int, int>> dirs = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        root = new TrieNode();

        for (auto& w : words) insert(w);

        int m = board.size(), n = board[0].size();
        vector<string> ans;

        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                int idx = board[i][j] - 'a';
                if (root->child[idx]) {           // ① 起點也要 Trie 有這個字母
                    char cache = board[i][j];
                    board[i][j] = '#';            // ② 起點先標 #
                    search(board, ans, i, j, root->child[idx]);
                    board[i][j] = cache;
                }
            }
        }

        return ans;
    }

    void insert(string& w) {
        TrieNode *cur = root;

        for (char& c : w) {
            int i = c - 'a';
            if (!cur->child[i]) cur->child[i] = new TrieNode();
            cur = cur->child[i];
        }
        cur->word = w;                    // 208 的 isWord 改成存字
    }

    void search(vector<vector<char>>& board, vector<string>& ans, int r, int c, TrieNode* node) {
        int m = board.size(), n = board[0].size();
        if (!node->word.empty()) {
            ans.push_back(node->word);        // ③ 收答案
            node->word.clear();               // ④ 清空,避免重複
        }

        for (auto& [dr, dc] : dirs) {
            int nr = r + dr, nc = c + dc;
            if (nr < 0 || nr >= m || nc < 0 || nc >= n) continue;
            if (board[nr][nc] == '#' || !node->child[board[nr][nc] - 'a']) continue;  // ⑤ 剪枝

            char cache = board[nr][nc];
            board[nr][nc] = '#';
            search(board, ans, nr, nc, node->child[cache - 'a']);
            board[nr][nc] = cache;            // ⑥ 還原
        }
    }
};

Follow-up 加速 · 剪掉找完的葉子// SHRINK THE TRIE AS YOU GO

上面的版本找到字只清空 word,節點還留著 —— 之後別的起點走到同一個前綴,照樣一路往下展開。 改成:每個節點記子節點數 cnt;回溯回來時,如果這個節點沒有子節點、也沒存字,就從父節點上拔掉。全部拔光(root->cnt == 0)就提早結束。

p212_prune.cpp C++17 
class TrieNode {
public:
    TrieNode *child[26] = { nullptr };
    string word = "";
    int cnt = 0;
};

class Solution {
public:
    TrieNode *root;
    int m, n;
    vector<string> ans;
    const vector<pair<int, int>> dirs = {{-1, 0}, {1, 0}, {0, -1}, {0, 1}};

    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        root = new TrieNode();
        for (auto& w : words) insert(w);

        m = board.size(), n = board[0].size();
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (root->cnt == 0) return ans;
                dfs(board, i, j, root);
            }
        }
        return ans;
    }

    void insert(string& w) {
        TrieNode *cur = root;
        for (char& c : w) {
            int i = c - 'a';
            if (!cur->child[i]) {
                cur->child[i] = new TrieNode();
                cur->cnt++;
            }
            cur = cur->child[i];
        }
        cur->word = w;
    }

    void dfs(vector<vector<char>>& board, int r, int c, TrieNode* parent) {
        if (r < 0 || r >= m || c < 0 || c >= n) return;
        char ch = board[r][c];
        if (ch == '#') return;

        TrieNode *node = parent->child[ch - 'a'];
        if (!node) return;

        if (!node->word.empty()) {
            ans.push_back(node->word);
            node->word.clear();
        }

        board[r][c] = '#';
        for (auto& [dr, dc] : dirs) dfs(board, r + dr, c + dc, node);
        board[r][c] = ch;

        if (node->cnt == 0 && node->word.empty()) {
            delete node;
            parent->child[ch - 'a'] = nullptr;
            parent->cnt--;
        }
    }
};

什麼時候有用、什麼時候沒用

有用:字真的被找到時。全 a 那組,第一個起點就把 a … a10 一路收完,整條鏈被拔光、root->cnt 歸零,第二格都不用看。 沒用:字找不到時,沒有葉子可以拔 —— aaaaaaaaab 那條鏈永遠留著,呼叫次數一模一樣,反而因為每個方向都多呼叫一次再判斷出界 / null 而變慢。 隨機字:3 萬個字只找到 612 個,幾乎沒得剪,時間主要花在建 Trie。 結論:面試先寫原版;被問到「很多字都找得到時怎麼加速」再提剪葉子,並說清楚它只在找得到時有效。
