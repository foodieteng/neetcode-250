完整實作// WALK · CREATE IF MISSING · CHECK END

p208.cpp C++17 
class TrieNode {
public:
    TrieNode* child[26] = { nullptr };   // ① 26 格全部 nullptr
    bool isWord = false;                 // ② 有字在這裡結束嗎
};

class Trie {
private:
    TrieNode* root;

public:
    Trie() {
        root = new TrieNode();
    }

    void insert(string word) {
        TrieNode* cur = root;
        for (char c : word) {
            int i = c - 'a';
            if (cur->child[i] == nullptr) {    // ③ 缺才建,已有就沿用
                cur->child[i] = new TrieNode();
            }
            cur = cur->child[i];
        }
        cur->isWord = true;                  // ④ 終點標記
    }

    bool search(string word, bool prefix = false) {
        TrieNode* cur = root;
        for (char &c : word) {
            int i = c - 'a';
            if (cur->child[i] == nullptr) {
                return false;                // 半路斷了
            }
            cur = cur->child[i];
        }
        if (prefix) return true;             // ⑤ 前綴:走得完就好
        return cur->isWord;                  // ⑥ 整字:還要 isWord
    }

    bool startsWith(string prefix) {
        return search(prefix, true);
    }
};

① ② 的初始化為什麼不能省// DEFAULT MEMBER INITIALIZERS

寫法實測
= { nullptr } / = false + new TrieNode()0 / 20,000 錯(本頁寫法)
有 ① ②,但 new TrieNode 沒有括號0 / 20,000 錯:預設成員初始化一樣會跑
拿掉 ① ② 的初始值,new TrieNode 沒括號未定義行為:新程序單跑剛好 0 錯;同一程序連跑 20,000 組直接 segfault;堆疊被弄髒時 19,234 / 20,000 錯(7,579 次當掉)
{ nullptr } 只寫了第一格,其餘 25 格會自動補 0,所以 26 格都是 nullptr。 沒初始化的 child[] 是記憶體裡剩下的垃圾,被當成指標就會亂跳。「本機跑過」不代表對 —— valgrind 會直接指出 cur->child[i] == nullptr 讀到未初始化的值。
