完整實作// NULL GUARD · LETTER · WILDCARD

p211.cpp C++17 
class TrieNode {
public:
    TrieNode* child[26] = { nullptr };
    bool isWord = false;
};

class WordDictionary {
public:
    TrieNode* root;

    WordDictionary() {
        root = new TrieNode();
    }

    void addWord(string word) {                          // 和 208 的 insert 一樣
        TrieNode *cur = root;

        for (char &c : word) {
            int i = c - 'a';
            if (!cur->child[i]) cur->child[i] = new TrieNode();
            cur = cur->child[i];
        }
        cur->isWord = true;
    }

    bool search(string word) {
        return dfs(word, 0, root);
    }

    bool dfs(const string& word, int idx, TrieNode* node) {
        if (!node) return false;                         // ① 兩種 null 都在這裡擋
        if (idx == word.size()) return node->isWord;     // ② 配完:看 isWord

        char c = word[idx];
        if (c != '.') return dfs(word, idx + 1, node->child[c - 'a']);   // ③ 字母:走一條

        for (TrieNode* child : node->child)              // ④ '.':26 條都試
            if (dfs(word, idx + 1, child)) return true;  // ⑤ 任一 true 就停
        return false;
    }
};

Follow-up 加速 · 節點池// ALLOCATION IS THE BOTTLENECK

瓶頸不在 search,在每建一個節點就 new 一次 216 B。改成一開始開好整塊陣列、用編號當指標;順便把字母改成迴圈走、'.' 迴圈跳過空格。

p211_pool.cpp C++17 
class WordDictionary {
private:
    static const int MAXN = 250001;   // 10^4 次 × 長度 25 + root
    int nxt[MAXN][26];                // nxt[node][c] = 子節點編號,0 = 不存在
    bool isWord[MAXN];
    int cnt = 1;                      // 0 號是 root

    bool dfs(const string& word, int idx, int node) {
        for (; idx < word.size() && word[idx] != '.'; idx++) {
            node = nxt[node][word[idx] - 'a'];
            if (!node) return false;
        }
        if (idx == word.size()) return isWord[node];
        for (int c = 0; c < 26; c++) {
            int child = nxt[node][c];
            if (child && dfs(word, idx + 1, child)) return true;
        }
        return false;
    }

public:
    WordDictionary() {
        memset(nxt[0], 0, sizeof(nxt[0]));
        isWord[0] = false;
    }

    void addWord(string word) {
        int cur = 0;
        for (char c : word) {
            int i = c - 'a';
            if (!nxt[cur][i]) {
                memset(nxt[cnt], 0, sizeof(nxt[cnt]));
                isWord[cnt] = false;
                nxt[cur][i] = cnt++;
            }
            cur = nxt[cur][i];
        }
        isWord[cur] = true;
    }

    bool search(string word) {
        return dfs(word, 0, 0);
    }
};

為什麼快、要注意什麼

快:少了幾十萬次小 new;節點排在連續記憶體,cache 好。只有建新節點時才 memset 那一列,不會每次清掉 25 萬列。 只改搜尋端沒用:迴圈走字母 + 跳過 null,但保留 new,實測三輪都落在 167–191 ms,和指標版分不出差別。 注意:物件本身 26,250,112 B(約 25 MB),LeetCode 用 new 建在 heap 沒事,當區域變數會爆 stack;MAXN 是從「104 次 × 長度 25 + root」推來的,換題要重算。
