完整實作// THE CODE

class Solution {
public:
    int splitArray(vector<int>& nums, int k) {
        // 值域:最小的候選是「最大元素自己一段」,最大的候選是「整條不切」
        int left = *max_element(nums.begin(), nums.end());
        int right = accumulate(nums.begin(), nums.end(), 0);

        while (left < right) {
            int mid = left + (right - left) / 2;

            if (countSegments(nums, mid) <= k) {
                right = mid;          // 可行 → 保留 mid,繼續往左壓
            } else {
                left = mid + 1;      // 不可行 → mid 已被否證,必須跨過
            }
        }

        return left;                // 第一個可行上限;答案必存在,免驗證
    }

private:
    // 上限為 limit 時,貪心「能塞就塞」所需的最少段數
    int countSegments(vector<int>& nums, int limit) {
        int cur = 0, segments = 1;

        for (int x : nums) {
            if (cur + x > limit) {
                cur = 0;           // 切一刀:歸零再放,不是「放了才歸零」
                segments++;
            }

            cur += x;
        }

        return segments;
    }
};

check 那三行的順序// RESET BEFORE ADD

countSegments 最容易寫錯的是「切刀」那一段。正確的語意是: 發現放不下 → 先把當前段結掉(cur = 0、段數 +1)→ 再把這個元素放進新的一段。 所以 cur += x 必須在 if 外面,兩條路徑共用。

// 正確:cur 歸零後才加 x —— x 進的是新段 if (cur + x > limit) { cur = 0; segments++; } cur += x; nums=[7,2,5,10,8], limit=15 x=7 7 <= 15 cur=7 x=2 9 <= 15 cur=9 x=5 14 <= 15 cur=14 x=10 24 > 15 切刀 → cur=0, segments=2, 然後 cur=10 x=8 18 > 15 切刀 → cur=0, segments=3, 然後 cur=8 // segments = 3
// 錯法:忘了 cur = 0 —— cur 變成前綴和,段數會虛胖 if (cur + x > limit) { segments++; } // 少了 cur = 0 cur += x; 實測最小反例:nums=[7,9,5], k=2 → 答 16(正解 14) 變體全掃:56000 組裡錯 16176 組 // cur 一路累加不歸零 ⇒ 幾乎每個元素都觸發切刀 ⇒ 段數暴增 ⇒ 二分被推到太大的 v

另外一個細節:因為 left 從 max(nums) 起跳,limit ≥ 每一個 x 恆成立, 所以「切刀之後 cur = x 仍然 ≤ limit」永遠不會失敗 —— 不需要額外檢查單一元素塞不塞得下。 左界的選擇同時保證了 check 的正確性。

// LOOP: while (left < right) 為什麼不會死迴圈 mid = left + (right - left) / 2 ⇒ left ≤ mid < right(因為 left < right,除法向下取整) 可行分支 right = mid ⇒ right 嚴格變小(mid < 原 right) 不可行分支 left = mid + 1 ⇒ left 嚴格變大 // 兩條路徑都讓區間嚴格縮短 ⇒ 必定收斂到 left == right
// EXIT: 為什麼 return left 不用再驗一次 right 這一側從初始就是「已知可行」的值(Σnums 必定可行,check = 1 ≤ k) 每次 right = mid 都是在 mid 被驗證可行之後才做 ⇒ 迴圈全程 right 恆為可行 ⇒ 收斂時 left == right 也可行 // 不變量:right 永遠指向一個可行解。這就是免驗證的理由
