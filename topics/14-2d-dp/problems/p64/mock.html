The 2-D version: start cell, two prefix-sum loops for the edges, then min(up, left) + the cell cost.

class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        vector<vector<int>> dp(m, vector<int>(n, INT_MAX));

        dp[0][0] = grid[0][0];
        for (int i = 1; i < n; i++)  // first row: prefix sums
            dp[0][i] = dp[0][i - 1] + grid[0][i];
        for (int i = 1; i < m; i++)  // first column: prefix sums
            dp[i][0] = dp[i - 1][0] + grid[i][0];

        for (int i = 1; i < m; i++)
            for (int j = 1; j < n; j++)
                dp[i][j] = min(dp[i - 1][j], dp[i][j - 1]) + grid[i][j];

        return dp[m - 1][n - 1];
    }
};

And the O(n)-space rolling version: one row, seeded with the first row's prefix sums.

class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m = grid.size(), n = grid[0].size();
        vector<int> dp(n, 0);
        dp[0] = grid[0][0];
        for (int j = 1; j < n; j++) dp[j] = dp[j - 1] + grid[0][j]; // first row

        for (int i = 1; i < m; i++) {
            dp[0] += grid[i][0];                    // first column: only "up"
            for (int j = 1; j < n; j++)
                dp[j] = min(dp[j], dp[j - 1]) + grid[i][j]; // dp[j]=up, dp[j-1]=left
        }
        return dp[n - 1];
    }
};

Both are O(mn) time. One thing worth flagging — in the rolling version, each new row starts with dp[0] += grid[i][0], because the first column has no left neighbor; its only predecessor is the value above, which dp[0] still holds.
